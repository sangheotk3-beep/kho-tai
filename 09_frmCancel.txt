' Quet QR (hoac go so lo / so phieu roi Enter):
'   QR hang - lo chua xuat -> 入庫取消 (tru ton kho)
'   QR hang - lo da xuat   -> 出庫取消 (tra ve ton kho, ca cac lo cung
'                             phieu + cung mat hang)
'   QR phieu - dong chua xuat -> 伝票取消
'   QR phieu - lo da xuat     -> 出庫取消 (tra ve ton kho)
' Quet chi them vao danh sach. Bam 取消実行 moi ghi vao sheet.
' Khong xoa dong nao: doi 区分 -> 入庫取消 / 出庫取消, trang thai -> 取消,
' ghi lich su vao CancelLog.
'==================================================

    Set cmdExec = UI_Button(Me, "cmdExec", "取消実行", 440, 516, 120, 34, BTN_DANGER)
    Set cmdClear = UI_Button(Me, "cmdClear", "クリア", 568, 516, 104, 34, BTN_SECONDARY)
    Set cmdClose = UI_Button(Me, "cmdClose", "閉じる", 680, 516, 104, 34, BTN_SECONDARY)

    UpdateCount
    UI_Msg lblMsg, "QRを読むと一覧に追加されます。「取消実行」で確定します。", MSG_INFO
    UI_Status Me, "行をダブルクリックすると一覧から外します。"

End Sub

Private Sub txtQR_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If KeyCode = 13 Then
        KeyCode = 0
        HandleScan
    End If
End Sub

Private Sub lstData_DblClick(ByVal Cancel As MSForms.ReturnBoolean)

    If lstData.ListIndex < 0 Then Exit Sub

    lstData.RemoveItem lstData.ListIndex
    UpdateCount
    UI_Msg lblMsg, "1 行を一覧から外しました。", MSG_INFO
    FocusQR

End Sub

Private Sub cmdClear_Click()
    lstData.Clear
    UpdateCount
    UI_Msg lblMsg, "一覧をクリアしました。", MSG_INFO
    FocusQR
End Sub

Private Sub cmdClose_Click()

    If lstData.ListCount > 0 Then
        If MsgBox("一覧の内容は取消されずに消えます。閉じますか？", _
                  vbYesNo + vbQuestion, APP_NAME) <> vbYes Then Exit Sub
    End If

    Unload Me

End Sub

Private Sub cmdExec_Click()

    If lstData.ListCount = 0 Then
        UI_Msg lblMsg, "取消する行がありません。QR を読み取ってください。", MSG_ERROR
        FocusQR
        Exit Sub
    End If

    ExecuteAll
    FocusQR

End Sub

Private Sub HandleScan()

    Dim raw As String
    Dim attr As String, siz As String, lot As String

    raw = CleanText(CStr(txtQR.Value))
    txtQR.Value = vbNullString

    If raw = "" Then Exit Sub

    If ParseGoodsQR(raw, attr, siz, lot) Then
        ScanLot lot
    ElseIf SlipExists(raw) Then
        ScanSlip raw
    Else
        ScanLot raw
    End If

    UpdateCount
    FocusQR

End Sub

Private Sub ScanLot(ByVal lot As String)

    Dim sh As Worksheet
    Dim r As Long, lastRow As Long
    Dim inRow As Long, outRow As Long
    Dim kind As String
    Dim key As String
    Dim added As Long

    If lot = "" Then
        UI_Msg lblMsg, "ロット番号を取得できません。", MSG_ERROR
        Exit Sub
    End If

    Set sh = Sht(SH_TRANS)
    lastRow = LastRowOf(sh, "D")

    For r = 2 To lastRow
        If StrComp(Trim$(CStr(sh.Cells(r, "G").Value)), lot, vbTextCompare) = 0 Then
            kind = Trim$(CStr(sh.Cells(r, "D").Value))
            If kind = K_IN Then inRow = r
            If kind = K_OUT Then outRow = r
        End If
    Next r

    If outRow > 0 Then

        key = MakeKey(sh.Cells(outRow, "B").Value, sh.Cells(outRow, "E").Value, sh.Cells(outRow, "F").Value)

        For r = 2 To lastRow
            If Trim$(CStr(sh.Cells(r, "D").Value)) = K_OUT Then
                If MakeKey(sh.Cells(r, "B").Value, sh.Cells(r, "E").Value, sh.Cells(r, "F").Value) = key Then
                    added = added + AddTransRow(sh, r, K_OUT_CANCEL)
                End If
            End If
        Next r

        If added = 0 Then
            UI_Msg lblMsg, "ロット " & lot & " はすでに一覧にあります。", MSG_ERROR
        Else
            UI_Msg lblMsg, "出庫済み → 在庫へ戻します（伝票 " & sh.Cells(outRow, "B").Value & _
                           " / " & added & " ロット追加）", MSG_OK
        End If

    ElseIf inRow > 0 Then

        If AddTransRow(sh, inRow, K_IN_CANCEL) = 0 Then
            UI_Msg lblMsg, "ロット " & lot & " はすでに一覧にあります。", MSG_ERROR
        Else
            UI_Msg lblMsg, "入庫取消: ロット " & lot & " を追加しました。", MSG_OK
        End If

    Else

        UI_Msg lblMsg, "「" & lot & "」は取消できるデータがありません（未登録 または 取消済み）。", MSG_ERROR

    End If

End Sub

Private Sub ScanSlip(ByVal slip As String)

    Dim shT As Worksheet
    Dim shD As Worksheet
    Dim r As Long
    Dim nSlip As Long, nShip As Long
    Dim k As Long

    Set shT = Sht(SH_TRANS)
    Set shD = Sht(SH_DETAIL)

    For r = 2 To LastRowOf(shD, "B")
        If StrComp(Trim$(CStr(shD.Cells(r, "B").Value)), slip, vbTextCompare) = 0 Then
            If Trim$(CStr(shD.Cells(r, "F").Value)) = ST_PENDING Then
                If Not InList("D" & r) Then
                    lstData.AddItem "D" & r
                    k = lstData.ListCount - 1
                    lstData.List(k, 1) = K_SLIP
                    lstData.List(k, 2) = CStr(shD.Cells(r, "B").Value)
                    lstData.List(k, 3) = "行番 " & CStr(shD.Cells(r, "J").Value)
                    lstData.List(k, 4) = CStr(shD.Cells(r, "C").Value)
                    lstData.List(k, 5) = CStr(shD.Cells(r, "D").Value)
                    lstData.List(k, 6) = CStr(shD.Cells(r, "E").Value)
                    nSlip = nSlip + 1
                End If
            End If
        End If
    Next r

    For r = 2 To LastRowOf(shT, "D")
        If Trim$(CStr(shT.Cells(r, "D").Value)) = K_OUT Then
            If StrComp(Trim$(CStr(shT.Cells(r, "B").Value)), slip, vbTextCompare) = 0 Then
                nShip = nShip + AddTransRow(shT, r, K_OUT_CANCEL)
            End If
        End If
    Next r

    If nSlip + nShip = 0 Then
        UI_Msg lblMsg, "伝票 " & slip & " は取消できる行がありません（取消済み または 一覧にあります）。", MSG_ERROR
    Else
        UI_Msg lblMsg, "伝票 " & slip & " : 未出庫 " & nSlip & " 行を取消 / 出庫済み " & _
                       nShip & " ロットを在庫へ戻します。", MSG_OK
    End If

End Sub

Private Function AddTransRow(ByVal sh As Worksheet, ByVal r As Long, ByVal kind As String) As Long

    Dim k As Long

    If InList("T" & r) Then Exit Function

    lstData.AddItem "T" & r
    k = lstData.ListCount - 1

    lstData.List(k, 1) = kind
    lstData.List(k, 2) = CStr(sh.Cells(r, "B").Value)
    lstData.List(k, 3) = CStr(sh.Cells(r, "G").Value)
    lstData.List(k, 4) = CStr(sh.Cells(r, "E").Value)
    lstData.List(k, 5) = CStr(sh.Cells(r, "F").Value)
    lstData.List(k, 6) = CStr(sh.Cells(r, "H").Value)

    lstData.TopIndex = k
    AddTransRow = 1

End Function

Private Function InList(ByVal code As String) As Boolean

    Dim i As Long

    For i = 0 To lstData.ListCount - 1
        If CStr(lstData.List(i, 0)) = code Then
            InList = True
            Exit Function
        End If
    Next i

End Function

' Phieu xuat (OutputDetails) hoac phieu 熊本行き (chi co trong 入出庫記録)
Private Function SlipExists(ByVal slip As String) As Boolean

    If Application.WorksheetFunction.CountIf(Sht(SH_DETAIL).Range("B:B"), slip) > 0 Then
        SlipExists = True
    ElseIf Application.WorksheetFunction.CountIf(Sht(SH_TRANS).Range("B:B"), slip) > 0 Then
        SlipExists = True
    End If

End Function

Private Function MakeKey(ByVal slip As Variant, ByVal attr As Variant, ByVal siz As Variant) As String
    MakeKey = LCase$(Trim$(CStr(slip))) & "|" & LCase$(Trim$(CStr(attr))) & "|" & LCase$(Trim$(CStr(siz)))
End Function

Private Sub ExecuteAll()

    Dim shT As Worksheet
    Dim shD As Worksheet
    Dim shipKeys As Object
    Dim i As Long, r As Long
    Dim code As String, kind As String
    Dim attr As String, siz As String, lot As String, slip As String
    Dim qty As Double
    Dim nIn As Long, nSlip As Long, nShip As Long
    Dim okN As Long, ngN As Long
    Dim errMsg As String, ngList As String
    Dim msg As String
    Dim tag As String

    For i = 0 To lstData.ListCount - 1
        Select Case CStr(lstData.List(i, 1))
            Case K_IN_CANCEL: nIn = nIn + 1
            Case K_SLIP: nSlip = nSlip + 1
            Case K_OUT_CANCEL: nShip = nShip + 1
        End Select
    Next i

    msg = "次の内容で取消します。よろしいですか？" & vbCrLf & vbCrLf
    If nIn > 0 Then msg = msg & "・入庫取消: " & nIn & " ロット（在庫から引く）" & vbCrLf
    If nSlip > 0 Then msg = msg & "・伝票取消: " & nSlip & " 行" & vbCrLf
    If nShip > 0 Then msg = msg & "・出庫取消: " & nShip & " ロット（在庫へ戻す）" & vbCrLf

    If MsgBox(msg, vbYesNo + vbExclamation, APP_NAME) <> vbYes Then Exit Sub

    Set shT = Sht(SH_TRANS)
    Set shD = Sht(SH_DETAIL)
    Set shipKeys = CreateObject("Scripting.Dictionary")
    tag = "[取消 " & Format$(Date, "yyyy/mm/dd") & "]"

    Application.ScreenUpdating = False

    For i = 0 To lstData.ListCount - 1

        code = CStr(lstData.List(i, 0))
        kind = CStr(lstData.List(i, 1))
        r = CLng(Mid$(code, 2))
        errMsg = vbNullString

        Select Case kind

            Case K_IN_CANCEL

                lot = Trim$(CStr(shT.Cells(r, "G").Value))
                attr = Trim$(CStr(shT.Cells(r, "E").Value))
                siz = Trim$(CStr(shT.Cells(r, "F").Value))
                qty = NumOf(shT.Cells(r, "H").Value)

                If Trim$(CStr(shT.Cells(r, "D").Value)) <> K_IN Then
                    errMsg = "すでに取消済みです"
                ElseIf LotState(lot) = LOT_OUT Then
                    errMsg = "出庫済みです"
                Else
                    errMsg = AdjustStock(attr, siz, -qty, 0, True)
                End If

                If errMsg = "" Then
                    shT.Cells(r, "D").Value = K_IN_CANCEL
                    AppendNote shT.Cells(r, "I"), tag
                    WriteCancelLog K_IN_CANCEL, CStr(shT.Cells(r, "B").Value), lot, attr, siz, qty
                End If

            Case K_SLIP

                If Trim$(CStr(shD.Cells(r, "F").Value)) <> ST_PENDING Then
                    errMsg = "状態が " & shD.Cells(r, "F").Value & " です"
                Else
                    shD.Cells(r, "F").Value = ST_CANCEL
                    AppendNote shD.Cells(r, "G"), tag
                    WriteCancelLog K_SLIP, CStr(shD.Cells(r, "B").Value), _
                                   "行番 " & CStr(shD.Cells(r, "J").Value), _
                                   CStr(shD.Cells(r, "C").Value), CStr(shD.Cells(r, "D").Value), _
                                   NumOf(shD.Cells(r, "E").Value)
                End If

            Case K_OUT_CANCEL

                lot = Trim$(CStr(shT.Cells(r, "G").Value))
                slip = Trim$(CStr(shT.Cells(r, "B").Value))
                attr = Trim$(CStr(shT.Cells(r, "E").Value))
                siz = Trim$(CStr(shT.Cells(r, "F").Value))
                qty = NumOf(shT.Cells(r, "H").Value)

                If Trim$(CStr(shT.Cells(r, "D").Value)) <> K_OUT Then
                    errMsg = "すでに取消済みです"
                Else
                    errMsg = AdjustStock(attr, siz, 0, -qty)
                End If

                If errMsg = "" Then
                    shT.Cells(r, "D").Value = K_OUT_CANCEL
                    AppendNote shT.Cells(r, "I"), tag
                    WriteCancelLog K_OUT_CANCEL, slip, lot, attr, siz, qty
                    shipKeys(MakeKey(slip, attr, siz)) = 1
                End If

        End Select

        If errMsg = "" Then
            okN = okN + 1
        Else
            ngN = ngN + 1
            ngList = ngList & vbCrLf & "・" & kind & " " & lstData.List(i, 3) & " : " & errMsg
        End If

    Next i

    ' Dong phieu cua hang da tra ve kho -> 取消
    If shipKeys.Count > 0 Then
        For r = 2 To LastRowOf(shD, "B")
            If Trim$(CStr(shD.Cells(r, "F").Value)) = ST_SHIPPED Then
                If shipKeys.Exists(MakeKey(shD.Cells(r, "B").Value, shD.Cells(r, "C").Value, shD.Cells(r, "D").Value)) Then
                    shD.Cells(r, "F").Value = ST_CANCEL
                    AppendNote shD.Cells(r, "G"), tag
                End If
            End If
        Next r
    End If

    Application.ScreenUpdating = True
    InvalidateCaches

    lstData.Clear
    UpdateCount

    If ngN = 0 Then
        UI_Msg lblMsg, okN & " 件を取消しました。", MSG_OK
        UI_Sound SOUND_DONE
    Else
        UI_Msg lblMsg, okN & " 件を取消しました。" & ngN & " 件はできませんでした。", MSG_ERROR
        MsgBox "次の行は取消できませんでした:" & ngList, vbExclamation, APP_NAME
    End If

    UI_Status Me, "取消実行: " & Format$(Now, "hh:mm") & "　（CancelLog に記録しました）", MSG_OK

End Sub

Private Sub UpdateCount()

    Dim i As Long
    Dim nIn As Long, nSlip As Long, nShip As Long

    For i = 0 To lstData.ListCount - 1
        Select Case CStr(lstData.List(i, 1))
            Case K_IN_CANCEL: nIn = nIn + 1
            Case K_SLIP: nSlip = nSlip + 1
            Case K_OUT_CANCEL: nShip = nShip + 1
        End Select
    Next i

    lblCount.Caption = "取消予定 " & lstData.ListCount & " 件" & vbLf & _
                       "入庫取消 " & nIn & "　／　伝票取消 " & nSlip & "　／　出庫取消 " & nShip

End Sub
