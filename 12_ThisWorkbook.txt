Option Explicit

'==================================================
' ThisWorkbook  -  ストレート管理
' Mo file : sao luu, khoa sheet du lieu, mo menu (Ctrl+M = mo menu)
' Dong file: tu dong luu + sao luu
'==================================================

Private Sub Workbook_Open()

    On Error Resume Next

    BackupNow "open"
    ProtectDataSheets

    ThisWorkbook.Worksheets(SH_STOCK).Activate
    Application.OnKey "^m", "'" & ThisWorkbook.Name & "'!AppStart"
    Application.OnTime Now + TimeSerial(0, 0, 1), "'" & ThisWorkbook.Name & "'!AppStart"

End Sub

' Dong file -> tu dong luu (khong hoi) + sao luu
Private Sub Workbook_BeforeClose(Cancel As Boolean)

    On Error Resume Next

    Application.OnKey "^m"

    If Not ThisWorkbook.Saved And Not ThisWorkbook.ReadOnly Then
        Application.DisplayAlerts = False
        ThisWorkbook.Save
        Application.DisplayAlerts = True
        BackupNow "close"
    End If

End Sub
