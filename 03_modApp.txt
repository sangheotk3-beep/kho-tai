' 在庫一覧: chi hien hang con ton, sap xep 属性 -> サイズ
Public Sub 在庫一覧を開く()

    Dim sh As Worksheet
    Dim ev As Boolean

    Set sh = Sht(SH_STOCK)

    ev = Application.EnableEvents
    Application.EnableEvents = False

    RefreshStockView sh
    sh.Activate
    sh.Range("A1").Select

    Application.EnableEvents = ev

End Sub

' Goi tu sheet 在庫 moi khi mo sheet (Worksheet_Activate)
Public Sub StockSheetActivated()

    Dim ev As Boolean

    ev = Application.EnableEvents
    Application.EnableEvents = False
    On Error Resume Next

    RefreshStockView Sht(SH_STOCK)

    Application.EnableEvents = ev

End Sub

' Sap xep (con ton truoc -> 属性 -> サイズ) va an hang co 在庫 <= 0
Public Sub RefreshStockView(ByVal sh As Worksheet)

    Dim lastRow As Long
    Dim r As Long
    Dim su As Boolean

    su = Application.ScreenUpdating
    Application.ScreenUpdating = False

    If sh.AutoFilterMode Then sh.AutoFilterMode = False

    lastRow = LastRowOf(sh, "A")

    If lastRow >= 3 Then

        ' Cot tam K: 0 = con ton, 1 = het ton
        For r = 2 To lastRow
            If NumOf(sh.Cells(r, "H").Value) > 0 Then
                sh.Cells(r, "K").Value = 0
            Else
                sh.Cells(r, "K").Value = 1
            End If
        Next r

        With sh.Sort
            .SortFields.Clear
            .SortFields.Add Key:=sh.Range("K2:K" & lastRow), Order:=xlAscending
            .SortFields.Add Key:=sh.Range("A2:A" & lastRow), Order:=xlAscending
            .SortFields.Add Key:=sh.Range("B2:B" & lastRow), Order:=xlAscending
            .SetRange sh.Range("A2:K" & lastRow)
            .Header = xlNo
            .Apply
        End With

        sh.Range("K2:K" & lastRow).ClearContents
        InvalidateStockCache

    End If

    If lastRow >= 2 Then
        sh.Range("A1:H" & lastRow).AutoFilter Field:=8, Criteria1:=">0"
    End If

    Application.ScreenUpdating = su

End Sub

Public Sub 品目マスタを開く()

Private Sub ShowSheetAtEnd(ByVal sheetName As String, ByVal col As String)

    Dim sh As Worksheet

    Set sh = Sht(sheetName)
    sh.Visible = xlSheetVisible
    sh.Activate

    ' Nhay toi dong cuoi (du lieu moi nhat)
    sh.Cells(NextRowOf(sh, col), 1).Select

End Sub

'==================================================
' 棚卸表 : 属性 | サイズ | 数量  (hang con ton)
'==================================================

Public Sub 棚卸表を印刷()

    Const FIRST_ROW As Long = 4

    Dim shS As Worksheet
    Dim sh As Worksheet
    Dim r As Long
    Dim outR As Long
    Dim qty As Double

    Set shS = Sht(SH_STOCK)
    Set sh = GetOrCreateSheet(SH_TANA)

    sh.Visible = xlSheetVisible

    Application.ScreenUpdating = False

    If sh.AutoFilterMode Then sh.AutoFilterMode = False
    sh.Cells.Clear
    sh.Cells.Font.Name = UI_FONT

    With sh.Range("A1:C1")
        .Merge
        .Value = "棚卸表"
        .Font.Size = 18
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
    End With

    sh.Range("A2:C2").Merge
    sh.Range("A2").Value = APP_NAME & "　　" & Format$(Date, "yyyy/mm/dd")
    sh.Range("A2").HorizontalAlignment = xlRight
    sh.Range("A2").Font.Size = 9

    sh.Range("A3:C3").Value = Array("属性", "サイズ", "数量")

    With sh.Range("A3:C3")
        .Font.Bold = True
        .Font.Color = CLR_WHITE
        .Interior.Color = CLR_PRIMARY
        .HorizontalAlignment = xlCenter
    End With

    outR = FIRST_ROW - 1

    For r = 2 To LastRowOf(shS, "A")
        qty = NumOf(shS.Cells(r, "H").Value)
        If qty > 0 Then
            outR = outR + 1
            sh.Cells(outR, "A").Value = shS.Cells(r, "A").Value
            sh.Cells(outR, "B").Value = shS.Cells(r, "B").Value
            sh.Cells(outR, "C").Value = qty
        End If
    Next r

    If outR < FIRST_ROW Then
        Application.ScreenUpdating = True
        sh.Visible = xlSheetHidden
        MsgBox "在庫のある品目がありません。", vbInformation, APP_NAME
        Exit Sub
    End If

    With sh.Sort
        .SortFields.Clear
        .SortFields.Add Key:=sh.Range("A" & FIRST_ROW & ":A" & outR), Order:=xlAscending
        .SortFields.Add Key:=sh.Range("B" & FIRST_ROW & ":B" & outR), Order:=xlAscending
        .SetRange sh.Range("A" & FIRST_ROW & ":C" & outR)
        .Header = xlNo
        .Apply
    End With

    With sh.Range("A3:C" & outR)
        .Font.Size = 11
        .RowHeight = 20
        .VerticalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous
        .Borders.Color = CLR_BORDER
    End With

    sh.Range("C" & FIRST_ROW & ":C" & outR).NumberFormat = "#,##0.##"

    sh.Columns("A").ColumnWidth = 18
    sh.Columns("B").ColumnWidth = 24
    sh.Columns("C").ColumnWidth = 12

    On Error Resume Next
    With sh.PageSetup
        .PrintArea = "$A$1:$C$" & outR
        .PrintTitleRows = "$3:$3"
        .Orientation = xlPortrait
        .PaperSize = xlPaperA4
        .Zoom = False
        .FitToPagesWide = 1
        .FitToPagesTall = False
        .CenterHorizontally = True
        .CenterFooter = "&P / &N"
    End With
    On Error GoTo 0

    Application.ScreenUpdating = True

    sh.Activate
    sh.Range("A1").Select
    ' Chon may in rieng cho 棚卸表 (neu co dat), xong tra lai may in cu
    Dim oldPrinter As String
    oldPrinter = Application.ActivePrinter
    If TANA_PRINTER <> "" Then
        On Error Resume Next
        Application.ActivePrinter = TANA_PRINTER
        If Err.Number <> 0 Then
            MsgBox "プリンター「" & TANA_PRINTER & "」が見つかりません。" & vbCrLf & _
                   "通常のプリンターで印刷します。", vbExclamation, APP_NAME
        End If
        On Error GoTo 0
    End If

    sh.PrintPreview

    On Error Resume Next
    Application.ActivePrinter = oldPrinter
    On Error GoTo 0

    ' In xong: an sheet, quay ve 在庫
    sh.Visible = xlSheetHidden
    On Error Resume Next
    Sht(SH_STOCK).Activate

End Sub

Public Function CountStockItems() As Long

    Dim sh As Worksheet
    Dim r As Long

    Set sh = Sht(SH_STOCK)

    For r = 2 To LastRowOf(sh, "A")
        If NumOf(sh.Cells(r, "H").Value) > 0 Then CountStockItems = CountStockItems + 1
    Next r

End Function

Public Function CountPendingSlips() As Long

    Dim sh As Worksheet
    Dim r As Long
    Dim d As Object
    Dim k As String

    Set sh = Sht(SH_DETAIL)
    Set d = CreateObject("Scripting.Dictionary")

    For r = 2 To LastRowOf(sh, "B")
        If Trim$(CStr(sh.Cells(r, "F").Value)) = ST_PENDING Then
            k = LCase$(Trim$(CStr(sh.Cells(r, "B").Value)))
            If Not d.Exists(k) Then d.Add k, 1
        End If
    Next r

    CountPendingSlips = d.Count

End Function

Public Function TodayShippedQty() As Double

    Dim sh As Worksheet
    Dim r As Long

    Set sh = Sht(SH_TRANS)

    For r = 2 To LastRowOf(sh, "D")
        If Trim$(CStr(sh.Cells(r, "D").Value)) = K_OUT Then
            If IsDate(sh.Cells(r, "C").Value) Then
                If Int(CDate(sh.Cells(r, "C").Value)) = Int(Date) Then
                    TodayShippedQty = TodayShippedQty + NumOf(sh.Cells(r, "H").Value)
                End If
            End If
        End If
    Next r

End Function

'   canh file, giu toi da BACKUP_KEEP ban moi nhat.

Public Sub ProtectDataSheets()

    Dim sh As Worksheet

    On Error Resume Next

    For Each sh In ThisWorkbook.Worksheets
        If IsLockedSheet(sh.Name) Then
            sh.Protect Password:=PROTECT_PW, UserInterfaceOnly:=True, _
                       AllowFiltering:=True, AllowFormattingColumns:=True
        End If
    Next sh

End Sub

' Khoa tat ca sheet tru sheet in (棚卸表 / 熊本行き)
Public Function IsLockedSheet(ByVal sheetName As String) As Boolean
    Select Case sheetName
        Case SH_HOME, SH_TANA, SH_KUMA
            IsLockedSheet = False
        Case Else
            IsLockedSheet = True
    End Select
End Function

' Mo khoa de sua tay (quan ly). Xong viec chay シート保護 hoac mo lai file.
Public Sub シート保護解除()

    Dim pw As String
    Dim sh As Worksheet

    If PROTECT_PW <> "" Then
        pw = InputBox("パスワードを入力してください。", APP_NAME)
        If pw <> PROTECT_PW Then
            MsgBox "パスワードが違います。", vbExclamation, APP_NAME
            Exit Sub
        End If
    End If

    On Error Resume Next
    For Each sh In ThisWorkbook.Worksheets
        If IsLockedSheet(sh.Name) Then sh.Unprotect Password:=PROTECT_PW
    Next sh

    MsgBox "シートの保護を解除しました。" & vbCrLf & _
           "作業が終わったら「シート保護」を実行するか、ファイルを開き直してください。", _
           vbInformation, APP_NAME

End Sub

Public Sub UnprotectDataSheetsSilent()

    Dim sh As Worksheet

    On Error Resume Next
    For Each sh In ThisWorkbook.Worksheets
        If IsLockedSheet(sh.Name) Then sh.Unprotect Password:=PROTECT_PW
    Next sh

End Sub

Public Sub シート保護()
    ProtectDataSheets
    MsgBox "シートを保護しました。", vbInformation, APP_NAME
End Sub

' Luu 1 ban sao vao thu muc backup
Public Sub BackupNow(ByVal tag As String)

    Dim folder As String
    Dim fname As String
    Dim ext As String

    On Error GoTo Done

    If ThisWorkbook.Path = "" Then Exit Sub

    folder = ThisWorkbook.Path & Application.PathSeparator & "backup"
    If Dir(folder, vbDirectory) = "" Then MkDir folder

    ext = Mid$(ThisWorkbook.Name, InStrRev(ThisWorkbook.Name, "."))
    fname = folder & Application.PathSeparator & "ストレート管理_" & _
            Format$(Now, "yyyymmdd_hhnnss") & "_" & tag & ext

    ThisWorkbook.SaveCopyAs fname

    CleanupBackups folder

Done:
End Sub

' Chi giu BACKUP_KEEP file moi nhat
Private Sub CleanupBackups(ByVal folder As String)

    Dim files() As String
    Dim n As Long
    Dim f As String
    Dim i As Long, j As Long
    Dim t As String

    f = Dir(folder & Application.PathSeparator & "ストレート管理_*.*")

    Do While f <> ""
        ReDim Preserve files(0 To n)
        files(n) = f
        n = n + 1
        f = Dir()
    Loop

    If n <= BACKUP_KEEP Then Exit Sub

    ' Sap xep ten (ten co ngay gio -> cu truoc)
    For i = 0 To n - 2
        For j = i + 1 To n - 1
            If files(j) < files(i) Then
                t = files(i): files(i) = files(j): files(j) = t
            End If
        Next j
    Next i

    On Error Resume Next
    For i = 0 To n - BACKUP_KEEP - 1
        Kill folder & Application.PathSeparator & files(i)
    Next i

End Sub

Public Sub バックアップフォルダを開く()

    Dim folder As String

    folder = ThisWorkbook.Path & Application.PathSeparator & "backup"
    If Dir(folder, vbDirectory) = "" Then
        MsgBox "まだバックアップがありません。", vbInformation, APP_NAME
        Exit Sub
    End If

    Shell "explorer.exe """ & folder & """", vbNormalFocus

End Sub
