' modData  -  ストレート管理  du lieu dung chung
'
' Cau truc sheet (giu nguyen nhu file cu):
'   マスタ        : A No | B 属性 | C サイズ | D 単位 | E 1QR本数 | F 期首在庫
'   在庫          : A 属性 | B サイズ | C 単位 | D 1QR本数 |
'                   E 期首在庫 | F 入庫 | G 出庫 | H 在庫 (=E+F-G)
'   入出庫記録    : A No | B 伝票番号 | C 日付 | D 区分 | E 属性 | F サイズ |
'                   G ロット番号 | H 数量 | I 備考 | J 状態
'   OutputDetails : A No | B 伝票番号 | C 属性 | D サイズ | E 数量 | F 状態 |
'                   G 備考 | H 出庫予定日 | I 出庫日 | J 行番
'   CancelLog     : A 日時 | B 種類 | C 伝票番号 | D ロット/行番 | E 属性 |
'                   F サイズ | G 本数
'
' QR hang (cach nhau dau phay): parts(5)=属性, parts(7)=サイズ, parts(15)=ロット
'==================================================

Public Function CleanText(ByVal s As String) As String
    s = Replace(s, vbCr, "")
    s = Replace(s, vbLf, "")
    s = Replace(s, vbTab, "")
    CleanText = Trim$(s)
End Function

' Ma phieu tu QR phieu (bo ky tu xuong dong may quet gui kem)
Public Function QR伝票コード(ByVal s As String) As String
    QR伝票コード = CleanText(s)
End Function

Public Function IsGoodsQR(ByVal raw As String) As Boolean
    IsGoodsQR = (UBound(Split(raw, ",")) >= 15)
End Function

' Tach QR hang. Tra ve False neu khong phai QR hang
Public Function ParseGoodsQR( _
    ByVal raw As String, _
    ByRef attr As String, ByRef siz As String, ByRef lot As String) As Boolean

    Dim parts() As String

    If InStr(1, raw, ",") = 0 Then Exit Function

    parts = Split(raw, ",")
    If UBound(parts) < 15 Then Exit Function

    attr = Trim$(parts(5))
    siz = Trim$(parts(7))
    lot = Trim$(parts(15))

    ParseGoodsQR = True

End Function

Public Function NumOf(ByVal v As Variant) As Double
    If IsNumeric(v) And Trim$(CStr(v)) <> "" Then NumOf = CDbl(v)
End Function

' So -> chuoi (dau cham thap phan, Val doc lai dung)
Public Function Num(ByVal v As Double) As String
    If Abs(v - Fix(v)) < EPS Then
        Num = CStr(CLng(v))
    Else
        Num = Replace(Format$(v, "0.###"), ",", ".")
    End If
End Function

Public Function FmtDate(ByVal v As Variant) As String
    If IsDate(v) Then
        FmtDate = Format$(CDate(v), "yyyy/mm/dd")
    Else
        FmtDate = CStr(v)
    End If
End Function

' Chuyen so toan goc va ky hieu ~ - ve dang thuong
Public Function NormalizeLotText(ByVal s As String) As String

    Dim i As Long

    For i = 0 To 9
        s = Replace(s, ChrW(&HFF10 + i), CStr(i))
    Next i

    s = Replace(s, ChrW(&HFF5E), "~")    ' fullwidth tilde
    s = Replace(s, ChrW(&H301C), "~")    ' wave dash
    s = Replace(s, ChrW(&HFF0D), "-")    ' fullwidth minus
    s = Replace(s, ChrW(&H2212), "-")    ' minus sign
    s = Replace(s, ChrW(&H2010), "-")    ' hyphen
    s = Replace(s, ChrW(&H2013), "-")    ' en dash
    s = Replace(s, ChrW(&H30FC), "-")    ' katakana long mark

    NormalizeLotText = Trim$(s)

End Function

' Lay don vi + so ong / 1 QR. False neu khong co hoac so ong <= 0
Public Function GetMasterInfo( _
    ByVal attr As String, ByVal siz As String, _
    ByRef unitName As String, ByRef unitsPerQR As Double) As Boolean

    Dim k As String
    Dim v As Variant

    EnsureMasterCache
    k = ItemKey(attr, siz)

    If mMaster.Exists(k) Then
        v = mMaster(k)
        unitName = CStr(v(0))
        unitsPerQR = CDbl(v(1))
        GetMasterInfo = (unitsPerQR > 0)
    End If

End Function

Public Function GetUnitsPerQR( _
    ByVal attr As String, ByVal siz As String, ByRef units As Double) As Boolean

    Dim u As String
    GetUnitsPerQR = GetMasterInfo(attr, siz, u, units)

End Function

Public Function FindStockRow(ByVal attr As String, ByVal siz As String) As Long

    Dim k As String
    Dim sh As Worksheet
    Dim r As Long

    EnsureStockCache
    k = ItemKey(attr, siz)

    If mStock.Exists(k) Then
        r = mStock(k)
        ' Kiem tra lai (phong khi dong bi doi cho)
        Set sh = Sht(SH_STOCK)
        If ItemKey(CStr(sh.Cells(r, "A").Value), CStr(sh.Cells(r, "B").Value)) = k Then
            FindStockRow = r
            Exit Function
        End If
        InvalidateCaches
        EnsureStockCache
        If mStock.Exists(k) Then FindStockRow = mStock(k)
    End If

End Function

Public Function StockOf(ByVal attr As String, ByVal siz As String) As Double

    Dim r As Long

    r = FindStockRow(attr, siz)
    If r > 0 Then StockOf = NumOf(Sht(SH_STOCK).Cells(r, "H").Value)

End Function

' Cong them vao 入庫(F) / 出庫(G) va tinh lai 在庫(H) = E + F - G
' createIfMissing: tao dong moi neu chua co (dung khi nhap kho)
' checkNeg: khong cho ton kho am
' Tra ve "" neu OK, nguoc lai la thong bao loi
Public Function AdjustStock( _
    ByVal attr As String, ByVal siz As String, _
    ByVal dIn As Double, ByVal dOut As Double, _
    Optional ByVal checkNeg As Boolean = False, _
    Optional ByVal createIfMissing As Boolean = False) As String

    Dim sh As Worksheet
    Dim r As Long
    Dim e As Double, f As Double, g As Double, h As Double
    Dim unitName As String
    Dim units As Double

    Set sh = Sht(SH_STOCK)
    r = FindStockRow(attr, siz)

    If r = 0 Then

        If Not createIfMissing Then
            AdjustStock = "在庫シートに品目がありません（" & attr & " / " & siz & "）"
            Exit Function
        End If

        GetMasterInfo attr, siz, unitName, units

        r = NextRowOf(sh, "A")
        sh.Cells(r, "A").Value = attr
        sh.Cells(r, "B").Value = siz
        InvalidateStockCache
        sh.Cells(r, "C").Value = unitName
        sh.Cells(r, "D").Value = units
        sh.Cells(r, "E").Value = 0
        sh.Cells(r, "F").Value = 0
        sh.Cells(r, "G").Value = 0
        sh.Cells(r, "H").Value = 0

    End If

    e = NumOf(sh.Cells(r, "E").Value)
    f = NumOf(sh.Cells(r, "F").Value) + dIn
    g = NumOf(sh.Cells(r, "G").Value) + dOut
    h = e + f - g

    If checkNeg And h < -EPS Then
        AdjustStock = "在庫が足りません（" & attr & " / " & siz & "　在庫 " & _
                      Num(NumOf(sh.Cells(r, "H").Value)) & " 本）"
        Exit Function
    End If

    sh.Cells(r, "E").Value = e
    sh.Cells(r, "F").Value = f
    sh.Cells(r, "G").Value = g
    sh.Cells(r, "H").Value = h

End Function

' Trang thai lo trong 入出庫記録 (bo qua dong da 取消).
' LOT_IN: tra ve attr / siz / qty cua dong 入庫
Public Function LotState( _
    ByVal lot As String, _
    Optional ByRef attr As String, _
    Optional ByRef siz As String, _
    Optional ByRef qty As Double) As Long

    Dim v As Variant

    If Trim$(lot) = "" Then Exit Function

    EnsureLotCache

    If mLots.Exists(LCase$(Trim$(lot))) Then
        v = mLots(LCase$(Trim$(lot)))
        LotState = CLng(v(0))
        If LotState = LOT_IN Then
            attr = CStr(v(1))
            siz = CStr(v(2))
            qty = CDbl(v(3))
        End If
    End If

End Function

' Lo da duoc dung (nhap hoac xuat, chua huy) -> khong cho nhap trung
Public Function LotUsed(ByVal lot As String) As Boolean
    LotUsed = (LotState(lot) <> LOT_NONE)
End Function

Public Function WriteTrans( _
    ByVal slip As String, ByVal d As Date, ByVal kind As String, _
    ByVal attr As String, ByVal siz As String, ByVal lot As String, _
    ByVal qty As Double, ByVal note As String) As Long

    Dim sh As Worksheet
    Dim r As Long

    Set sh = Sht(SH_TRANS)
    r = NextRowOf(sh, "D")

    sh.Cells(r, "A").Value = r - 1
    sh.Cells(r, "B").NumberFormat = "@"
    sh.Cells(r, "B").Value = slip
    sh.Cells(r, "C").NumberFormat = "yyyy/mm/dd"
    sh.Cells(r, "C").Value = d
    sh.Cells(r, "D").Value = kind
    sh.Cells(r, "E").Value = attr
    sh.Cells(r, "F").Value = siz
    sh.Cells(r, "G").NumberFormat = "@"
    sh.Cells(r, "G").Value = lot
    sh.Cells(r, "H").Value = qty
    sh.Cells(r, "I").Value = note
    sh.Cells(r, "J").Value = "完了"

    CacheLotWrite lot, kind, attr, siz, qty, r

    WriteTrans = r

End Function

Public Sub WriteCancelLog( _
    ByVal kind As String, ByVal slip As String, ByVal ref As String, _
    ByVal attr As String, ByVal siz As String, ByVal qty As Double)

    Dim sh As Worksheet
    Dim r As Long

    Set sh = Sht(SH_LOG)
    r = NextRowOf(sh, "A")

    sh.Cells(r, "A").NumberFormat = "yyyy/mm/dd hh:mm"
    sh.Cells(r, "A").Value = Now
    sh.Cells(r, "B").Value = kind
    sh.Cells(r, "C").NumberFormat = "@"
    sh.Cells(r, "C").Value = slip
    sh.Cells(r, "D").NumberFormat = "@"
    sh.Cells(r, "D").Value = ref
    sh.Cells(r, "E").Value = attr
    sh.Cells(r, "F").Value = siz
    sh.Cells(r, "G").Value = qty

End Sub

'==================================================
' 期首在庫 (マスタ cot F -> 在庫 cot E)
'==================================================

' Ghi 期首在庫 cua 1 dong マスタ vao 在庫 (tao dong 在庫 neu chua co)
Public Function SyncOpeningRow(ByVal r As Long) As Boolean

    Dim shM As Worksheet
    Dim shS As Worksheet
    Dim attr As String, siz As String
    Dim v As Variant
    Dim opening As Double
    Dim sr As Long

    Set shM = Sht(SH_MASTER)
    Set shS = Sht(SH_STOCK)

    attr = Trim$(CStr(shM.Cells(r, "B").Value))
    siz = Trim$(CStr(shM.Cells(r, "C").Value))
    v = shM.Cells(r, "F").Value

    If attr = "" Or siz = "" Then Exit Function

    If Trim$(CStr(v)) = "" Then
        opening = 0
    ElseIf IsNumeric(v) Then
        opening = CDbl(v)
    Else
        Exit Function
    End If

    sr = FindStockRow(attr, siz)

    If sr = 0 Then
        If opening = 0 Then Exit Function
        sr = NextRowOf(shS, "A")
        shS.Cells(sr, "A").Value = attr
        shS.Cells(sr, "B").Value = siz
        InvalidateStockCache
        shS.Cells(sr, "F").Value = 0
        shS.Cells(sr, "G").Value = 0
    End If

    shS.Cells(sr, "C").Value = shM.Cells(r, "D").Value
    shS.Cells(sr, "D").Value = shM.Cells(r, "E").Value
    shS.Cells(sr, "E").Value = opening
    shS.Cells(sr, "H").Value = opening + NumOf(shS.Cells(sr, "F").Value) - NumOf(shS.Cells(sr, "G").Value)

    SyncOpeningRow = True

End Function

' Goi tu sheet マスタ khi sua o (chi xu ly cot F)
Public Sub MasterChanged(ByVal Target As Range)

    Dim sh As Worksheet
    Dim rng As Range
    Dim cell As Range
    Dim lastRow As Long

    InvalidateCaches

    Set sh = Target.Worksheet
    lastRow = LastRowOf(sh, "B")
    If lastRow < 2 Then Exit Sub

    Set rng = Application.Intersect(Target, sh.Range("F2:F" & lastRow))
    If rng Is Nothing Then Exit Sub

    Application.EnableEvents = False
    On Error GoTo Done

    For Each cell In rng.Cells
        SyncOpeningRow cell.Row
    Next cell

Done:
    Application.EnableEvents = True

End Sub

' Nut 在庫へ反映 tren sheet マスタ: ghi toan bo cot F
Public Sub 期首在庫を在庫へ反映()

    Dim sh As Worksheet
    Dim r As Long
    Dim n As Long

    Set sh = Sht(SH_MASTER)

    If MsgBox("マスタの「期首在庫」を在庫シートへ反映します。" & vbCrLf & _
              "（在庫 = 期首在庫 + 入庫 - 出庫 で再計算します）" & vbCrLf & vbCrLf & _
              "よろしいですか？", vbYesNo + vbQuestion, "ストレート管理") <> vbYes Then Exit Sub

    Application.ScreenUpdating = False

    For r = 2 To LastRowOf(sh, "B")
        If Trim$(CStr(sh.Cells(r, "F").Value)) <> "" Then
            If SyncOpeningRow(r) Then n = n + 1
        End If
    Next r

    Application.ScreenUpdating = True

    MsgBox n & " 品目の期首在庫を反映しました。", vbInformation, "ストレート管理"

End Sub

' 在庫 cot E -> マスタ cot F (chi dien o con trong)
Public Sub FillMasterOpeningFromStock()

    Dim shM As Worksheet
    Dim shS As Worksheet
    Dim r As Long
    Dim sr As Long
    Dim ev As Boolean

    Set shM = Sht(SH_MASTER)
    Set shS = Sht(SH_STOCK)

    ev = Application.EnableEvents
    Application.EnableEvents = False

    For r = 2 To LastRowOf(shM, "B")
        If Trim$(CStr(shM.Cells(r, "F").Value)) = "" Then
            sr = FindStockRow(CStr(shM.Cells(r, "B").Value), CStr(shM.Cells(r, "C").Value))
            If sr > 0 Then
                If NumOf(shS.Cells(sr, "E").Value) <> 0 Then
                    shM.Cells(r, "F").Value = NumOf(shS.Cells(sr, "E").Value)
                End If
            End If
        End If
    Next r

    Application.EnableEvents = ev

End Sub

Public Sub InvalidateCaches()
    Set mMaster = Nothing
    Set mStock = Nothing
    Set mLots = Nothing
End Sub

Public Sub InvalidateStockCache()
    Set mStock = Nothing
End Sub

Private Function ItemKey(ByVal attr As String, ByVal siz As String) As String
    ItemKey = LCase$(Trim$(attr)) & vbTab & LCase$(Trim$(siz))
End Function

Private Sub EnsureMasterCache()

    Dim sh As Worksheet
    Dim n As Long
    Dim a As Variant
    Dim i As Long
    Dim k As String

    Set sh = Sht(SH_MASTER)
    n = LastRowOf(sh, "B")

    If Not mMaster Is Nothing And n = mMasterRows Then Exit Sub

    Set mMaster = CreateObject("Scripting.Dictionary")
    mMasterRows = n
    If n < 2 Then Exit Sub

    a = sh.Range("B2:E" & n).Value

    For i = 1 To UBound(a, 1)
        k = ItemKey(CStr(a(i, 1)), CStr(a(i, 2)))
        If Not mMaster.Exists(k) Then
            mMaster.Add k, Array(Trim$(CStr(a(i, 3))), NumOf(a(i, 4)))
        End If
    Next i

End Sub

Private Sub EnsureStockCache()

    Dim sh As Worksheet
    Dim n As Long
    Dim a As Variant
    Dim i As Long
    Dim k As String

    Set sh = Sht(SH_STOCK)
    n = LastRowOf(sh, "A")

    If Not mStock Is Nothing And n = mStockRows Then Exit Sub

    Set mStock = CreateObject("Scripting.Dictionary")
    mStockRows = n
    If n < 2 Then Exit Sub

    a = sh.Range("A2:B" & n).Value

    For i = 1 To UBound(a, 1)
        k = ItemKey(CStr(a(i, 1)), CStr(a(i, 2)))
        If Not mStock.Exists(k) Then mStock.Add k, i + 1
    Next i

End Sub

Private Sub EnsureLotCache()

    Dim sh As Worksheet
    Dim n As Long
    Dim a As Variant
    Dim i As Long

    Set sh = Sht(SH_TRANS)
    n = LastRowOf(sh, "D")
    If LastRowOf(sh, "G") > n Then n = LastRowOf(sh, "G")

    If Not mLots Is Nothing And n = mLotRows Then Exit Sub

    Set mLots = CreateObject("Scripting.Dictionary")
    mLotRows = n
    If n < 2 Then Exit Sub

    ' D 区分 | E 属性 | F サイズ | G ロット | H 数量
    a = sh.Range("D2:H" & n).Value

    For i = 1 To UBound(a, 1)
        AddLotToCache CStr(a(i, 4)), Trim$(CStr(a(i, 1))), _
                      Trim$(CStr(a(i, 2))), Trim$(CStr(a(i, 3))), NumOf(a(i, 5))
    Next i

End Sub

' Quy tac giong LotState cu: co 出庫 -> LOT_OUT; co 入庫 -> LOT_IN; 取消 bo qua
Private Sub AddLotToCache( _
    ByVal lot As String, ByVal kind As String, _
    ByVal attr As String, ByVal siz As String, ByVal qty As Double)

    Dim k As String

    k = LCase$(Trim$(lot))
    If k = "" Then Exit Sub

    If kind = K_OUT Then
        mLots(k) = Array(LOT_OUT, "", "", 0#)
    ElseIf kind = K_IN Then
        If mLots.Exists(k) Then
            If mLots(k)(0) = LOT_OUT Then Exit Sub
        End If
        mLots(k) = Array(LOT_IN, attr, siz, qty)
    End If

End Sub

' Goi tu WriteTrans: cap nhat cache ngay, khong can doc lai sheet
Private Sub CacheLotWrite( _
    ByVal lot As String, ByVal kind As String, _
    ByVal attr As String, ByVal siz As String, _
    ByVal qty As Double, ByVal r As Long)

    If mLots Is Nothing Then Exit Sub
    If r <> mLotRows + 1 Then
        Set mLots = Nothing
        Exit Sub
    End If

    AddLotToCache lot, kind, attr, siz, qty
    mLotRows = r

End Sub

' Doc file van ban theo ma (shift_jis / utf-8)
Public Function ReadTextFile(ByVal path As String, ByVal charset As String) As String

    Dim st As Object

    Set st = CreateObject("ADODB.Stream")
    st.Type = 2                 ' text
    st.Charset = charset
    st.Open
    st.LoadFromFile path
    ReadTextFile = st.ReadText(-1)
    st.Close

End Function

' Tach CSV thanh cac dong; moi dong la mang chuoi (0-based). Ho tro "..." co dau phay.
Public Function ParseCsv(ByVal txt As String, ByVal delim As String) As Collection

    Dim csvRows As New Collection
    Dim fields() As String
    Dim nF As Long
    Dim cur As String
    Dim inQ As Boolean
    Dim i As Long
    Dim ch As String

    txt = Replace(txt, vbCrLf, vbLf)
    txt = Replace(txt, vbCr, vbLf)
    If Right$(txt, 1) <> vbLf Then txt = txt & vbLf

    ReDim fields(0 To 0)
    nF = 0

    For i = 1 To Len(txt)

        ch = Mid$(txt, i, 1)

        If inQ Then
            If ch = """" Then
                If Mid$(txt, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQ = False
                End If
            Else
                cur = cur & ch
            End If
        Else
            Select Case ch
                Case """"
                    inQ = True
                Case delim
                    ReDim Preserve fields(0 To nF)
                    fields(nF) = Trim$(cur)
                    nF = nF + 1
                    cur = vbNullString
                Case vbLf
                    ReDim Preserve fields(0 To nF)
                    fields(nF) = Trim$(cur)
                    csvRows.Add fields
                    nF = 0
                    cur = vbNullString
                    ReDim fields(0 To 0)
                Case Else
                    cur = cur & ch
            End Select
        End If

    Next i

    Set ParseCsv = csvRows

End Function

' Lay o thu col (1-based) cua 1 dong CSV. col <= 0 hoac khong co -> ""
Public Function CsvField(ByVal rw As Variant, ByVal col As Long) As String
    If col <= 0 Then Exit Function
    If col - 1 > UBound(rw) Then Exit Function
    CsvField = Trim$(CStr(rw(col - 1)))
End Function
