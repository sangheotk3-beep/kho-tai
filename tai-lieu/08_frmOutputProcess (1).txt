'==================================================
' frmOutputProcess  -  実際の出庫処理
'
' 1. Quet QR phieu (伝票) -> hien cac dong 未出庫 cua phieu
' 2. Quet QR hang hoac go so lo (hang cu khong QR)
'    - So ong moi lo = 1QR本数 (マスタ)
'    - Lo co QR phai da 入庫. Lo da xuat -> loi
'    - Dang doc 1 mat hang thi khoa mat hang do den khi du
'    - Hang cu: chon dong trong danh sach roi go so lo
'    - Khong vuot so luong dong, khong vuot ton kho
' 3. Doc du tat ca dong -> tu dong ghi 出庫 (入出庫記録), tru 在庫,
'    OutputDetails -> 出庫済
' - Khong duoc doc phieu khac khi dang doc 1 phieu
' - 取消: huy toan bo phan dang doc (chua ghi gi)
'==================================================

Private Sub ResetSlipState()

    mSlip = vbNullString
    mNote = vbNullString
    mN = 0
    mS = 0
    mCur = -1

    Erase mRow: Erase mLine: Erase mAttr: Erase mSize: Erase mNeed: Erase mDone
    Erase sLot: Erase sAttr: Erase sSize: Erase sQty: Erase sOld

    txtSlipCode.Value = vbNullString
    lblProgress.Caption = vbNullString

End Sub

Private Sub ShowError(ByVal msg As String)
    UI_Msg lblMsg, msg, MSG_ERROR

End Sub

Private Function ShipDateText() As String
    If IsDate(txtDate.Value) Then
        ShipDateText = Format$(CDate(txtDate.Value), "yyyy/mm/dd")
    Else
        ShipDateText = Format$(Date, "yyyy/mm/dd")
    End If
End Function

Private Sub ProcessInput()

    Dim raw As String
    Dim code As String

    raw = CleanText(CStr(txtQR.Value))
    If raw = vbNullString Then Exit Sub

    If mSlip = "" Then

        If IsGoodsQR(raw) Then
            ShowError "先に伝票QRコードを読み取ってください。"
            Exit Sub
        End If

        LoadSlip raw

    Else

        code = QR伝票コード(raw)

        If Not IsGoodsQR(raw) And SlipExists(code) Then
            If StrComp(code, mSlip, vbTextCompare) = 0 Then
                ShowError "同じ伝票を読み取り中です。商品QRまたはロット番号を読み取ってください。"
            Else
                ShowError "別の伝票は読み取れません（処理中: " & mSlip & "）。先に完了するか「取消」を押してください。"
            End If
            Exit Sub
        End If

        ProcessLot raw

    End If

End Sub

Private Sub LoadSlip(ByVal raw As String)

    Dim sh As Worksheet
    Dim r As Long
    Dim slip As String
    Dim found As Boolean
    Dim needQty As Double

    If Not IsDate(txtDate.Value) Then
        ShowError "出庫日を正しく入力してください（例 2026/10/31）。"
        Exit Sub
    End If

    slip = QR伝票コード(raw)
    If slip = "" Then FocusQR: Exit Sub

    Set sh = Sht(SH_DETAIL)
    mN = 0

    For r = 2 To LastRowOf(sh, "B")

        If StrComp(Trim$(CStr(sh.Cells(r, "B").Value)), slip, vbTextCompare) = 0 Then

            found = True

            If Trim$(CStr(sh.Cells(r, "F").Value)) = ST_PENDING Then

                needQty = NumOf(sh.Cells(r, "E").Value)

                If needQty > 0 Then

                    ReDim Preserve mRow(0 To mN)
                    ReDim Preserve mLine(0 To mN)
                    ReDim Preserve mAttr(0 To mN)
                    ReDim Preserve mSize(0 To mN)
                    ReDim Preserve mNeed(0 To mN)
                    ReDim Preserve mDone(0 To mN)

                    mRow(mN) = r
                    mLine(mN) = LineNoOf(sh, r, slip)
                    mAttr(mN) = Trim$(CStr(sh.Cells(r, "C").Value))
                    mSize(mN) = Trim$(CStr(sh.Cells(r, "D").Value))
                    mNeed(mN) = needQty
                    mDone(mN) = 0

                    If mNote = "" Then mNote = CStr(sh.Cells(r, "G").Value)

                    mN = mN + 1

                End If

            End If

        End If

    Next r

    If Not found Then
        mN = 0
        ShowError "この伝票は登録されていません: " & slip
        Exit Sub
    End If

    If mN = 0 Then
        ShowError "この伝票は既に出庫済み（または取消）です: " & slip
        Exit Sub
    End If

    mSlip = slip
    mCur = -1
    mS = 0

    txtSlipCode.Value = slip

    ShowSlipList
    UpdateProgress
    UI_Msg lblMsg, "伝票を読み込みました（" & mN & " 行）。商品QRまたはロット番号を読み取ってください。", MSG_OK
    FocusQR

End Sub

' 行番: cot J; neu trong thi dem so dong cung phieu tu tren xuong
Private Function LineNoOf(ByVal sh As Worksheet, ByVal r As Long, ByVal slip As String) As Long

    Dim k As Long

    If IsNumeric(sh.Cells(r, "J").Value) And Trim$(CStr(sh.Cells(r, "J").Value)) <> "" Then
        LineNoOf = CLng(sh.Cells(r, "J").Value)
        Exit Function
    End If

    For k = 2 To r
        If StrComp(Trim$(CStr(sh.Cells(k, "B").Value)), slip, vbTextCompare) = 0 Then
            LineNoOf = LineNoOf + 1
        End If
    Next k

End Function

Private Function SlipExists(ByVal slip As String) As Boolean
    If slip = "" Then Exit Function
    SlipExists = (Application.WorksheetFunction.CountIf(Sht(SH_DETAIL).Range("B:B"), slip) > 0)
End Function

' Dang doc phieu: chi hien dong con thieu
Private Sub ShowSlipList()

    Dim i As Long
    Dim k As Long
    Dim units As Double

    lstList.Clear

    For i = 0 To mN - 1

        If mDone(i) < mNeed(i) - EPS Then

            lstList.AddItem CStr(i)
            k = lstList.ListCount - 1

            lstList.List(k, 1) = ShipDateText()
            lstList.List(k, 2) = mSlip
            lstList.List(k, 3) = CStr(mLine(i))
            lstList.List(k, 4) = mAttr(i)
            lstList.List(k, 5) = mSize(i)

            If GetUnitsPerQR(mAttr(i), mSize(i), units) Then
                lstList.List(k, 6) = Num(units)
            Else
                lstList.List(k, 6) = "-"
            End If

            lstList.List(k, 7) = Num(mDone(i)) & " / " & Num(mNeed(i))

            If i = mCur Then lstList.ListIndex = k

        End If

    Next i

    lblInfo.Caption = "伝票 " & mSlip & "　残り " & lstList.ListCount & " 行"

End Sub

' Chua doc phieu: tat ca dong 未出庫 co ngay <= ngay xuat
Private Sub ShowPending()

    Dim sh As Worksheet
    Dim r As Long
    Dim k As Long
    Dim units As Double
    Dim shipDate As Date
    Dim d As Date
    Dim slip As String

    lstList.Clear

    If Not IsDate(txtDate.Value) Then Exit Sub
    shipDate = CDate(txtDate.Value)

    Set sh = Sht(SH_DETAIL)

    For r = 2 To LastRowOf(sh, "B")

        If Trim$(CStr(sh.Cells(r, "F").Value)) = ST_PENDING Then

            If IsDate(sh.Cells(r, "H").Value) Then
                d = CDate(sh.Cells(r, "H").Value)
            Else
                d = shipDate
            End If

            If Int(d) <= Int(shipDate) Then

                slip = Trim$(CStr(sh.Cells(r, "B").Value))

                lstList.AddItem CStr(r)
                k = lstList.ListCount - 1

                lstList.List(k, 1) = Format$(d, "yyyy/mm/dd")
                lstList.List(k, 2) = slip
                lstList.List(k, 3) = CStr(LineNoOf(sh, r, slip))
                lstList.List(k, 4) = Trim$(CStr(sh.Cells(r, "C").Value))
                lstList.List(k, 5) = Trim$(CStr(sh.Cells(r, "D").Value))

                If GetUnitsPerQR(CStr(lstList.List(k, 4)), CStr(lstList.List(k, 5)), units) Then
                    lstList.List(k, 6) = Num(units)
                Else
                    lstList.List(k, 6) = "-"
                End If

                lstList.List(k, 7) = "0 / " & Num(NumOf(sh.Cells(r, "E").Value))

            End If

        End If

    Next r

    lblInfo.Caption = ShipDateText() & " までの未出庫: " & lstList.ListCount & " 行"

End Sub

'==================================================
' TIEN DO
'==================================================

Private Sub UpdateProgress()

    Dim idx As Long

    If mSlip = "" Then
        lblProgress.Caption = vbNullString
        Exit Sub
    End If

    If mCur >= 0 Then
        idx = mCur
    ElseIf SelectedLine() >= 0 Then
        idx = SelectedLine()
    Else
        idx = OnlyOpenLine()
    End If

    If idx < 0 Then
        lblProgress.Caption = "読取待ち"
    Else
        lblProgress.Caption = mAttr(idx) & "  " & mSize(idx) & "　　" & _
                              Num(mDone(idx)) & " / " & Num(mNeed(idx)) & " 本"
    End If

End Sub

Private Function SelectedLine() As Long

    SelectedLine = -1
    If mSlip = "" Then Exit Function
    If lstList.ListIndex < 0 Then Exit Function

    SelectedLine = CLng(lstList.List(lstList.ListIndex, 0))

End Function

Private Function OnlyOpenLine() As Long

    Dim i As Long
    Dim cnt As Long
    Dim pos As Long

    OnlyOpenLine = -1
    pos = -1

    For i = 0 To mN - 1
        If mDone(i) < mNeed(i) - EPS Then
            cnt = cnt + 1
            pos = i
        End If
    Next i

    If cnt = 1 Then OnlyOpenLine = pos

End Function

' DOC LO
'==================================================

Private Sub ProcessLot(ByVal raw As String)

    Dim lot As String
    Dim attr As String, siz As String
    Dim qrAttr As String, qrSize As String
    Dim qty As Double
    Dim state As Long
    Dim isQR As Boolean
    Dim isOld As Boolean
    Dim idx As Long
    Dim i As Long
    Dim recAttr As String, recSize As String
    Dim recQty As Double

    isQR = ParseGoodsQR(raw, qrAttr, qrSize, lot)
    If Not isQR Then lot = raw

    If lot = "" Then
        ShowError "ロット番号を取得できません。"
        Exit Sub
    End If

    For i = 0 To mS - 1
        If StrComp(sLot(i), lot, vbTextCompare) = 0 Then
            ShowError "このロットは既に読み取り済みです: " & lot
            Exit Sub
        End If
    Next i

    state = LotState(lot, recAttr, recSize, recQty)

    If state = LOT_OUT Then
        ShowError "このロットは既に出庫済みです: " & lot
        Exit Sub
    End If

    If isQR And state <> LOT_IN Then
        ShowError "このロットは入庫登録されていません: " & lot
        Exit Sub
    End If

    If state = LOT_IN Then

        isOld = False
        attr = recAttr
        siz = recSize

        If mCur >= 0 Then

            If StrComp(mAttr(mCur), attr, vbTextCompare) <> 0 _
            Or StrComp(mSize(mCur), siz, vbTextCompare) <> 0 Then
                ShowError "別の品目です。読み取り中の品目を完了してください（読取中: " & _
                          mAttr(mCur) & " " & mSize(mCur) & " ／ 今回: " & attr & " " & siz & "）"
                Exit Sub
            End If

            idx = mCur

        Else

            idx = FindLineByItem(attr, siz)

            If idx < 0 Then
                ShowError "この伝票にない品目、または既に完了しています: " & attr & " " & siz
                Exit Sub
            End If

        End If

    Else

        ' Hang cu khong QR
        isOld = True

        If mCur >= 0 Then
            idx = mCur
        ElseIf SelectedLine() >= 0 Then
            idx = SelectedLine()
        Else
            idx = OnlyOpenLine()
        End If

        If idx < 0 Then
            ShowError "旧在庫のロットです。リストの明細（品目）を選択してから入力してください。"
            Exit Sub
        End If

        attr = mAttr(idx)
        siz = mSize(idx)

    End If

    ' So ong 1 lo = 1QR本数 trong マスタ
    If Not GetUnitsPerQR(attr, siz, qty) Then
        If isOld Or recQty <= 0 Then
            ShowError "マスタに 1QR本数 が登録されていません: " & attr & " / " & siz
            Exit Sub
        End If
        qty = recQty
    End If

    If mNeed(idx) - mDone(idx) < qty - EPS Then
        ShowError "数量オーバーです（" & mAttr(idx) & " " & mSize(idx) & "　必要 " & _
                  Num(mNeed(idx)) & " / 読取済 " & Num(mDone(idx)) & " / このロット " & Num(qty) & " 本）"
        Exit Sub
    End If

    If FindStockRow(attr, siz) = 0 Then
        ShowError "在庫シートに該当品目がありません: " & attr & " / " & siz
        Exit Sub
    End If

    If StockOf(attr, siz) < ScannedQty(attr, siz) + qty - EPS Then
        ShowError "在庫が不足しています（" & attr & " / " & siz & "　在庫 " & _
                  Num(StockOf(attr, siz)) & " 本）"
        Exit Sub
    End If

    ReDim Preserve sLot(0 To mS)
    ReDim Preserve sAttr(0 To mS)
    ReDim Preserve sSize(0 To mS)
    ReDim Preserve sQty(0 To mS)
    ReDim Preserve sOld(0 To mS)

    sLot(mS) = lot
    sAttr(mS) = attr
    sSize(mS) = siz
    sQty(mS) = qty
    sOld(mS) = isOld
    mS = mS + 1

    mDone(idx) = mDone(idx) + qty
    mCur = idx

    If mDone(idx) >= mNeed(idx) - EPS Then
        mCur = -1
        ShowSlipList
        UI_Msg lblMsg, attr & " " & siz & " : 読み取り完了", MSG_OK
    Else
        ShowSlipList
        If isOld Then
            UI_Msg lblMsg, "OK（旧在庫）  " & lot & "　" & Num(qty) & " 本", MSG_OK
        Else
            UI_Msg lblMsg, "OK  " & lot & "　" & Num(qty) & " 本", MSG_OK
        End If
    End If

    UpdateProgress

    If AllDone() Then
        CommitSlip
    Else
        FocusQR
    End If

End Sub

Private Function AllDone() As Boolean

    Dim i As Long

    For i = 0 To mN - 1
        If mDone(i) < mNeed(i) - EPS Then Exit Function
    Next i

    AllDone = True

End Function

Private Function FindLineByItem(ByVal attr As String, ByVal siz As String) As Long

    Dim i As Long

    FindLineByItem = -1

    For i = 0 To mN - 1
        If mDone(i) < mNeed(i) - EPS Then
            If StrComp(mAttr(i), attr, vbTextCompare) = 0 _
            And StrComp(mSize(i), siz, vbTextCompare) = 0 Then
                FindLineByItem = i
                Exit Function
            End If
        End If
    Next i

End Function

Private Function ScannedQty(ByVal attr As String, ByVal siz As String) As Double

    Dim i As Long

    For i = 0 To mS - 1
        If StrComp(sAttr(i), attr, vbTextCompare) = 0 _
        And StrComp(sSize(i), siz, vbTextCompare) = 0 Then
            ScannedQty = ScannedQty + sQty(i)
        End If
    Next i

End Function

Private Sub CommitSlip()

    Dim shDet As Worksheet
    Dim i As Long
    Dim shipDate As Date
    Dim doneSlip As String
    Dim lotCount As Long
    Dim total As Double
    Dim note As String
    Dim errMsg As String

    On Error GoTo CommitError

    shipDate = CDate(txtDate.Value)
    Set shDet = Sht(SH_DETAIL)

    Application.ScreenUpdating = False

    For i = 0 To mS - 1

        note = mNote
        If sOld(i) Then note = Trim$(mNote & " (旧在庫)")

        WriteTrans mSlip, shipDate, K_OUT, sAttr(i), sSize(i), sLot(i), sQty(i), note

        errMsg = AdjustStock(sAttr(i), sSize(i), 0, sQty(i))
        If errMsg <> "" Then Err.Raise vbObjectError + 1, , errMsg

        total = total + sQty(i)

    Next i

    For i = 0 To mN - 1
        shDet.Cells(mRow(i), "F").Value = ST_SHIPPED
        shDet.Cells(mRow(i), "I").NumberFormat = "yyyy/mm/dd"
        shDet.Cells(mRow(i), "I").Value = shipDate
    Next i

    Application.ScreenUpdating = True

    doneSlip = mSlip
    lotCount = mS

    ResetSlipState
    ShowPending

    UI_Msg lblMsg, "出庫完了: 伝票 " & doneSlip & "（" & lotCount & " ロット / " & Num(total) & " 本）", MSG_OK
    UI_Status Me, "出庫登録: " & doneSlip & "  " & Format$(Now, "hh:mm"), MSG_OK
    UI_Sound SOUND_DONE

    MsgBox "全て読み取り完了しました。出庫を登録しました。" & vbCrLf & vbCrLf & _
           "伝票: " & doneSlip & vbCrLf & _
           "ロット数: " & lotCount & vbCrLf & _
           "本数: " & Num(total) & " 本", vbInformation, APP_NAME

    FocusQR
    Exit Sub

CommitError:
    Application.ScreenUpdating = True
    MsgBox "出庫登録中にエラーが発生しました。" & vbCrLf & Err.Description & vbCrLf & _
           "入出庫記録・在庫・OutputDetails を確認してください。", vbCritical, APP_NAME

End Sub
