Public Sub UI_InitForm( _
    ByVal frm As Object, _
    ByVal title As String, _
    ByVal subTitle As String, _
    ByVal w As Single, _
    ByVal h As Single)

    Dim lbl As MSForms.Label

    frm.Caption = APP_NAME & "  -  " & title
    frm.BackColor = CLR_BG
    frm.StartUpPosition = 1

    frm.Width = w
    frm.Height = h
    frm.Width = w + (frm.Width - frm.InsideWidth)
    frm.Height = h + (frm.Height - frm.InsideHeight)

    ' Thanh tieu de
    Set lbl = UI_Label(frm, "uiHeader", "", 0, 0, w, HEADER_H, 10, False, CLR_WHITE)
    lbl.BackStyle = fmBackStyleOpaque
    lbl.BackColor = CLR_PRIMARY

    Set lbl = UI_Label(frm, "uiAccent", "", 0, HEADER_H, w, 3, 8, False, CLR_WHITE)
    lbl.BackStyle = fmBackStyleOpaque
    lbl.BackColor = CLR_ACCENT

    UI_Label frm, "uiTitle", title, 18, 8, w - 220, 26, 16, True, CLR_WHITE
    UI_Label frm, "uiSub", subTitle, 19, 34, w - 220, 16, 9, False, CLR_HEADSUB

    Set lbl = UI_Label(frm, "uiApp", APP_NAME, w - 200, 11, 182, 16, 9, True, CLR_HEADSUB)
    lbl.TextAlign = fmTextAlignRight

    Set lbl = UI_Label(frm, "uiDate", TodayText(), w - 200, 30, 182, 18, 11, False, CLR_WHITE)
    lbl.TextAlign = fmTextAlignRight

    ' Thanh trang thai
    Set lbl = UI_Label(frm, "uiStatusBar", "", 0, h - 24, w, 24, 9, False, CLR_MUTED)
    lbl.BackStyle = fmBackStyleOpaque
    lbl.BackColor = CLR_STATUSBG

    UI_Label frm, "lblStatus", "", 14, h - 19, w - 28, 16, 9, False, CLR_MUTED

End Sub

Public Function TodayText() As String
    TodayText = Format$(Date, "yyyy/mm/dd") & " (" & _
                Mid$("日月火水木金土", Weekday(Date), 1) & ")"
End Function

' Thanh trang thai duoi cung
Public Sub UI_Status( _
    ByVal frm As Object, _
    ByVal msg As String, _
    Optional ByVal kind As Long = MSG_INFO)

    On Error Resume Next

    With frm.Controls("lblStatus")
        .Caption = msg
        Select Case kind
            Case MSG_OK: .ForeColor = CLR_SUCCESS
            Case MSG_ERROR: .ForeColor = CLR_DANGER
            Case Else: .ForeColor = CLR_MUTED
        End Select
    End With

End Sub

' O thong bao co mau nen (xanh = OK, do = loi)
Public Sub UI_Msg( _
    ByVal lbl As MSForms.Label, _
    ByVal msg As String, _
    Optional ByVal kind As Long = MSG_INFO)

    lbl.Caption = "  " & msg
    lbl.BackStyle = fmBackStyleOpaque

    Select Case kind
        Case MSG_OK
            lbl.BackColor = CLR_SUCCESS_BG
            lbl.ForeColor = CLR_SUCCESS
            UI_Sound SOUND_OK
        Case MSG_ERROR
            lbl.BackColor = CLR_DANGER_BG
            lbl.ForeColor = CLR_DANGER
            UI_Sound SOUND_ERROR
        Case Else
            lbl.BackColor = CLR_INFO_BG
            lbl.ForeColor = CLR_PRIMARY
    End Select

End Sub

Public Function UI_Label( _
    ByVal frm As Object, ByVal nm As String, ByVal cap As String, _
    ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
    Optional ByVal fs As Single = 10, Optional ByVal isBold As Boolean = False, _
    Optional ByVal color As Long = CLR_TEXT) As MSForms.Label

    Dim lbl As MSForms.Label

    Set lbl = frm.Controls.Add("Forms.Label.1", nm)

    With lbl
        .Left = x: .Top = y: .Width = w: .Height = h
        .Caption = cap
        .Font.Name = UI_FONT
        .Font.Size = fs
        .Font.Bold = isBold
        .ForeColor = color
        .BackStyle = fmBackStyleTransparent
        .WordWrap = False
    End With

    Set UI_Label = lbl

End Function

' Listbox + hang tieu de cot.
' widths: mang do rong (pt). Cot rong 0 = an, khong ve tieu de.
Public Function UI_List( _
    ByVal frm As Object, ByVal nm As String, _
    ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
    ByVal widths As Variant, ByVal heads As Variant) As MSForms.ListBox

    Dim lst As MSForms.ListBox
    Dim lbl As MSForms.Label
    Dim i As Long
    Dim cx As Single
    Dim lastVis As Long
    Dim ws As String

    ' Nen hang tieu de
    Set lbl = UI_Label(frm, nm & "_hbar", "", x, y, w, 22)
    lbl.BackStyle = fmBackStyleOpaque
    lbl.BackColor = CLR_PRIMARY

    For i = LBound(widths) To UBound(widths)
        If widths(i) > 0 Then lastVis = i
    Next i

    cx = x + 3

    For i = LBound(widths) To UBound(widths)

        If widths(i) > 0 Then
            Set lbl = UI_Label(frm, nm & "_h" & i, " " & CStr(heads(i)), _
                               cx, y + 4, CSng(widths(i)), 16, 10, True, CLR_WHITE)
            If i = lastVis Then lbl.Width = x + w - cx
        End If

        cx = cx + widths(i)
        ws = ws & CStr(CLng(widths(i))) & " pt;"

    Next i

    Set lst = frm.Controls.Add("Forms.ListBox.1", nm)

    With lst
        .Left = x: .Top = y + 22: .Width = w: .Height = h - 22
        .Font.Name = UI_FONT
        .Font.Size = 11
        .ForeColor = CLR_TEXT
        .BackColor = CLR_WHITE
        .SpecialEffect = fmSpecialEffectFlat
        .BorderStyle = fmBorderStyleSingle
        .BorderColor = CLR_BORDER
        .ColumnCount = UBound(widths) - LBound(widths) + 1
        .ColumnWidths = Left$(ws, Len(ws) - 1)
        .ColumnHeads = False
        .IntegralHeight = False
    End With

    Set UI_List = lst

End Function

'==================================================
' AM THANH
' Dung file am thanh co san cua Windows (C:\Windows\Media).
' Muon doi tieng: sua ten file .wav ben duoi.
'==================================================

Public Sub UI_Sound(ByVal kind As Long)

    Dim f As String
    Dim sysAlias As String

    If Not SOUND_ON Then Exit Sub

    On Error GoTo Fallback

    Select Case kind
        Case SOUND_OK
            f = "Windows Navigation Start.wav"
            sysAlias = "SystemAsterisk"
        Case SOUND_ERROR
            f = "Windows Critical Stop.wav"
            sysAlias = "SystemHand"
        Case SOUND_DONE
            f = "tada.wav"
            sysAlias = "SystemExclamation"
        Case SOUND_WARN
            f = "Windows Exclamation.wav"
            sysAlias = "SystemQuestion"
        Case Else
            Exit Sub
    End Select

    f = Environ$("WINDIR") & "\Media\" & f

    If Dir(f) <> "" Then
        PlaySound f, 0, SND_ASYNC Or SND_FILENAME Or SND_NODEFAULT
    Else
        PlaySound sysAlias, 0, SND_ASYNC Or SND_ALIAS
    End If

    Exit Sub

Fallback:
    Beep

End Sub
