Private Sub cmdNextBox_Click()

    If BoxLotCount(mBox) = 0 Then

        FocusQR
        Exit Sub
    End If

    mBox = mBox + 1

End Sub

Private Function BoxLotCount(ByVal b As Long) As Long

    Dim i As Long

    For i = 0 To lstData.ListCount - 1
        If Val(lstData.List(i, 5)) = b Then BoxLotCount = BoxLotCount + 1
    Next i

End Function

' Danh sach thung co hang, theo thu tu -> so thu tu in (1, 2, ...)
Private Function BoxOrder() As Object

    Dim d As Object
    Dim i As Long, b As Long, maxB As Long, n As Long

    Set d = CreateObject("Scripting.Dictionary")

    For i = 0 To lstData.ListCount - 1
        If Val(lstData.List(i, 5)) > maxB Then maxB = Val(lstData.List(i, 5))
    Next i

    For b = 1 To maxB
        If BoxLotCount(b) > 0 Then
            n = n + 1
            d.Add CStr(b), n
        End If
    Next b

    Set BoxOrder = d

End Function

Private Sub cmdPrint_Click()

    Dim slip As String

    If lstData.ListCount = 0 Then
        UI_Msg lblMsg, "印刷する商品がありません。", MSG_ERROR
        FocusQR
        Exit Sub
    End If

    If MsgBox(BoxOrder().Count & " 箱 / " & lstData.ListCount & " ロット（" & lblQty.Caption & "）を印刷して出庫します。" & vbCrLf & _
              "在庫から差し引きます。よろしいですか？", _
              vbYesNo + vbQuestion, APP_NAME) <> vbYes Then
        FocusQR
        Exit Sub
    End If

    slip = "熊本-" & Format$(Now, "yyyymmdd-hhnnss")

    If Not BuildAndPrint(slip) Then
        If MsgBox("印刷できませんでした。出庫だけ確定しますか？" & vbCrLf & _

            Exit Sub
        End If
    End If

    If CommitShip(slip) Then
        lstData.Clear
        mBox = 1
        UpdateTotals
        LoadOldAttributes
        UI_Msg lblMsg, "出庫しました（" & slip & "）。", MSG_OK
        UI_Sound SOUND_DONE
        UI_Status Me, "熊本行き 出庫: " & slip, MSG_OK
    End If

    FocusQR

End Sub

Private Sub HandleScan()

    Dim raw As String
    Dim attr As String, siz As String, lot As String
    Dim units As Double
    Dim state As Long
    Dim isOld As Boolean
    Dim k As Long

    raw = CleanText(CStr(txtQR.Value))
    txtQR.Value = vbNullString

    If raw = "" Then Exit Sub

    If InStr(1, raw, ",") > 0 Then

        If Not ParseGoodsQR(raw, attr, siz, lot) Then
            UI_Msg lblMsg, "商品QRではありません。", MSG_ERROR
            FocusQR
            Exit Sub
        End If

        If lot = "" Then
            UI_Msg lblMsg, "ロット番号がありません。", MSG_ERROR
            FocusQR
            Exit Sub
        End If

        state = LotState(lot)
        If state = LOT_NONE Then isOld = True

    Else

        lot = raw
        state = LotState(lot, attr, siz)

        If state = LOT_NONE Then
            If lstData.ListIndex < 0 Then
                UI_Msg lblMsg, "ロット " & lot & " は登録がありません。QRのない旧在庫は下の「旧在庫」欄で属性・サイズを選んで追加してください。", MSG_ERROR
                FocusQR
                Exit Sub
            End If
            attr = CStr(lstData.List(lstData.ListIndex, 0))
            siz = CStr(lstData.List(lstData.ListIndex, 1))
            isOld = True
        End If

    End If

    If state = LOT_OUT Then
        UI_Msg lblMsg, "ロット " & lot & " は出庫済みです。", MSG_ERROR
        FocusQR
        Exit Sub
    End If

    AddEntry attr, siz, lot, isOld
    FocusQR

End Sub

' Kiem tra chung + them 1 lo vao danh sach
Private Function AddEntry( _
    ByVal attr As String, ByVal siz As String, _
    ByVal lot As String, ByVal isOld As Boolean) As Boolean

    Dim units As Double
    Dim k As Long

    If LotInList(lot) Then
        UI_Msg lblMsg, "ロット " & lot & " はもう読みました。", MSG_ERROR
        Exit Function
    End If

    If Not GetUnitsPerQR(attr, siz, units) Then
        UI_Msg lblMsg, "マスタに 1QR本数 がありません: " & attr & " / " & siz, MSG_ERROR
        Exit Function
    End If

    If StockOf(attr, siz) < ScannedQty(attr, siz) + units - EPS Then
        UI_Msg lblMsg, "在庫が足りません: " & attr & " / " & siz & "（在庫 " & Num(StockOf(attr, siz)) & " 本）", MSG_ERROR
        Exit Function
    End If

    lstData.AddItem attr
    k = lstData.ListCount - 1
    lstData.List(k, 1) = siz
    lstData.List(k, 2) = lot
    lstData.List(k, 3) = Num(units)
    lstData.List(k, 4) = IIf(isOld, "1", "")
    lstData.List(k, 5) = CStr(mBox)
    lstData.ListIndex = k

    UpdateTotals

    If isOld Then
        UI_Msg lblMsg, "OK（旧在庫）  " & attr & " / " & siz & " / " & lot & "（" & Num(units) & " 本）", MSG_OK
    Else
        UI_Msg lblMsg, "OK  " & attr & " / " & siz & " / " & lot & "（" & Num(units) & " 本）", MSG_OK
    End If

    AddEntry = True

End Function

Private Sub LoadOldAttributes()

    Dim sh As Worksheet
    Dim r As Long
    Dim s As String

    cboAttr.Clear
    cboSize.Clear

    Set sh = Sht(SH_STOCK)

    For r = 2 To LastRowOf(sh, "A")
        s = Trim$(CStr(sh.Cells(r, "A").Value))
        If s <> "" And NumOf(sh.Cells(r, "H").Value) > 0 Then
            If Not ComboHas(cboAttr, s) Then cboAttr.AddItem s
        End If
    Next r

End Sub

Private Sub cboAttr_Change()

    Dim sh As Worksheet
    Dim r As Long
    Dim attr As String
    Dim s As String

    cboSize.Clear
    attr = Trim$(CStr(cboAttr.Value))
    If attr = "" Then Exit Sub

    Set sh = Sht(SH_STOCK)

    For r = 2 To LastRowOf(sh, "A")
        If StrComp(Trim$(CStr(sh.Cells(r, "A").Value)), attr, vbTextCompare) = 0 _
        And NumOf(sh.Cells(r, "H").Value) > 0 Then
            s = Trim$(CStr(sh.Cells(r, "B").Value))
            If s <> "" And Not ComboHas(cboSize, s) Then cboSize.AddItem s
        End If
    Next r

End Sub

Private Sub cboSize_Change()
    On Error Resume Next
    If cboSize.ListIndex >= 0 Then txtLot.SetFocus
End Sub

Private Sub AddOld()

    Dim attr As String, siz As String, lot As String
    Dim recAttr As String, recSize As String
    Dim state As Long
    Dim isOld As Boolean

    attr = Trim$(CStr(cboAttr.Value))
    siz = Trim$(CStr(cboSize.Value))
    lot = NormalizeLotText(CStr(txtLot.Value))

    If attr = "" Or siz = "" Then
        UI_Msg lblMsg, "旧在庫: 属性とサイズを選択してください。", MSG_ERROR
        Exit Sub
    End If

    If lot = "" Then
        UI_Msg lblMsg, "旧在庫: ロット番号を入力してください。", MSG_ERROR
        txtLot.SetFocus
        Exit Sub
    End If

    state = LotState(lot, recAttr, recSize)

    If state = LOT_OUT Then
        UI_Msg lblMsg, "ロット " & lot & " は出庫済みです。", MSG_ERROR
        txtLot.SetFocus
        Exit Sub
    End If

    If state = LOT_IN Then
        ' Lo nay da 入庫 (co QR) -> phai dung mat hang
        If StrComp(recAttr, attr, vbTextCompare) <> 0 Or StrComp(recSize, siz, vbTextCompare) <> 0 Then
            UI_Msg lblMsg, "ロット " & lot & " は " & recAttr & " / " & recSize & " で入庫されています。", MSG_ERROR
            txtLot.SetFocus
            Exit Sub
        End If
        isOld = False
    Else
        isOld = True
    End If

    If AddEntry(attr, siz, lot, isOld) Then
        txtLot.Value = vbNullString
    End If

    On Error Resume Next
    txtLot.SetFocus

End Sub

Private Function LotInList(ByVal lot As String) As Boolean

    Dim i As Long

    For i = 0 To lstData.ListCount - 1
        If StrComp(CStr(lstData.List(i, 2)), lot, vbTextCompare) = 0 Then
            LotInList = True
            Exit Function
        End If
    Next i

End Function

Private Function ScannedQty(ByVal attr As String, ByVal siz As String) As Double

    Dim i As Long

    For i = 0 To lstData.ListCount - 1
        If StrComp(CStr(lstData.List(i, 0)), attr, vbTextCompare) = 0 _
        And StrComp(CStr(lstData.List(i, 1)), siz, vbTextCompare) = 0 Then
            ScannedQty = ScannedQty + Val(lstData.List(i, 3))
        End If
    Next i

End Function

Private Sub UpdateTotals()

    Dim i As Long
    Dim qty As Double
    Dim bQty As Double
    Dim bN As Long
    Dim nBox As Long

    For i = 0 To lstData.ListCount - 1
        qty = qty + Val(lstData.List(i, 3))
        If Val(lstData.List(i, 5)) = mBox Then
            bN = bN + 1
            bQty = bQty + Val(lstData.List(i, 3))
        End If
    Next i

    lblLots.Caption = CStr(lstData.ListCount)
    lblQty.Caption = Num(qty) & " 本"

    nBox = BoxOrder().Count
    If bN = 0 And mBox > nBox Then nBox = nBox + 1

    lblBox.Caption = "現在の箱: " & mBox & " 箱目　（" & bN & " ロット / " & Num(bQty) & " 本）　　全 " & _
                     IIf(nBox < 1, 1, nBox) & " 箱"

End Sub

Private Function CommitShip(ByVal slip As String) As Boolean

    Dim need As Object
    Dim i As Long
    Dim attr As String, siz As String, lot As String
    Dim qty As Double
    Dim key As Variant
    Dim note As String
    Dim errMsg As String

    Set need = CreateObject("Scripting.Dictionary")

    For i = 0 To lstData.ListCount - 1

        lot = CStr(lstData.List(i, 2))

        If LotState(lot) = LOT_OUT Then
            MsgBox "ロット " & lot & " はすでに出庫済みです。一覧から削除してください。", vbExclamation, APP_NAME
            Exit Function
        End If

        key = CStr(lstData.List(i, 0)) & vbTab & CStr(lstData.List(i, 1))
        need(key) = need(key) + Val(lstData.List(i, 3))

    Next i

    For Each key In need.keys
        attr = Split(key, vbTab)(0)
        siz = Split(key, vbTab)(1)
        If StockOf(attr, siz) < need(key) - EPS Then
            MsgBox "在庫が足りません: " & attr & " / " & siz & vbCrLf & _
                   "在庫 " & Num(StockOf(attr, siz)) & " 本 / 出庫 " & Num(need(key)) & " 本", _
                   vbExclamation, APP_NAME
            Exit Function
        End If
    Next key

    Dim boxes As Object
    Dim boxTag As String

    Set boxes = BoxOrder()

    On Error GoTo CommitError
    Application.ScreenUpdating = False

    For i = 0 To lstData.ListCount - 1

        boxTag = ""
        If boxes.Count > 1 Then boxTag = " 箱" & boxes(CStr(Val(lstData.List(i, 5)))) & "/" & boxes.Count

        attr = CStr(lstData.List(i, 0))
        siz = CStr(lstData.List(i, 1))
        lot = CStr(lstData.List(i, 2))
        qty = Val(lstData.List(i, 3))

        If CStr(lstData.List(i, 4)) = "1" Then
            note = "熊本行き" & boxTag & " (旧在庫)"
        Else
            note = "熊本行き" & boxTag
        End If

        WriteTrans slip, Date, K_OUT, attr, siz, lot, qty, note

        errMsg = AdjustStock(attr, siz, 0, qty)
        If errMsg <> "" Then Err.Raise vbObjectError + 1, , errMsg

    Next i

    Application.ScreenUpdating = True
    CommitShip = True
    Exit Function

CommitError:
    Application.ScreenUpdating = True
    MsgBox "出庫登録中にエラーが発生しました。" & vbCrLf & Err.Description, vbCritical, APP_NAME

End Function

Private Function BuildAndPrint(ByVal slip As String) As Boolean

    Dim sh As Worksheet
    Dim keys As Object
    Dim order As Collection
    Dim i As Long, g As Long
    Dim key As String
    Dim attr As String, siz As String, lots As String
    Dim qty As Double
    Dim nLot As Long
    Dim r As Long, topR As Long
    Dim bTop() As Long, bBot() As Long

    Dim boxes As Object
    Dim bk As Variant
    Dim boxNo As Long

    ' Gom theo 箱 -> 材質 + 寸法 (giu thu tu quet)
    Set boxes = BoxOrder()
    Set keys = CreateObject("Scripting.Dictionary")
    Set order = New Collection

    For Each bk In boxes.keys
        For i = 0 To lstData.ListCount - 1
            If CStr(Val(lstData.List(i, 5))) = bk Then
                key = bk & "|" & LCase$(lstData.List(i, 0)) & "|" & LCase$(lstData.List(i, 1))
                If Not keys.Exists(key) Then
                    keys.Add key, i
                    order.Add key
                End If
            End If
        Next i
    Next bk

    ' Chon may in truoc, tao sheet moi -> sheet lay dung kho giay cua may in do
    Dim oldPrinter As String
    Dim ev As Boolean
    oldPrinter = Application.ActivePrinter
    If LABEL_PRINTER <> "" Then
        On Error Resume Next
        Application.ActivePrinter = LABEL_PRINTER
        If Err.Number <> 0 Then
            MsgBox "プリンター「" & LABEL_PRINTER & "」が見つかりません。" & vbCrLf & _
                   "通常のプリンターで印刷します。", vbExclamation, APP_NAME
        End If
        On Error GoTo 0
    End If

    On Error Resume Next
    Application.DisplayAlerts = False
    ThisWorkbook.Worksheets(SH_KUMA).Visible = xlSheetVisible
    ThisWorkbook.Worksheets(SH_KUMA).Delete
    Application.DisplayAlerts = True
    On Error GoTo 0

    Set sh = GetOrCreateSheet(SH_KUMA)
    sh.Visible = xlSheetVisible

    Application.ScreenUpdating = False

    sh.ResetAllPageBreaks
    sh.Cells.Clear
    sh.Cells.Font.Name = UI_FONT
    sh.Columns("A").ColumnWidth = 16
    sh.Columns("B").ColumnWidth = 55

    sh.Range("A1").Value = "熊本行き"
    sh.Range("A1").Font.Size = 14
    sh.Range("A1").Font.Bold = True
    sh.Range("B1").Value = Format$(Date, "yyyy/mm/dd") & "　　No. " & slip
    sh.Range("B1").HorizontalAlignment = xlRight
    sh.Range("B1").Font.Size = 10

    r = 3
    ReDim bTop(1 To order.Count)
    ReDim bBot(1 To order.Count)

    For g = 1 To order.Count

        key = order(g)
        lots = vbNullString
        qty = 0: nLot = 0

        For i = 0 To lstData.ListCount - 1
            If CStr(Val(lstData.List(i, 5))) & "|" & LCase$(lstData.List(i, 0)) & "|" & LCase$(lstData.List(i, 1)) = key Then
                boxNo = boxes(CStr(Val(lstData.List(i, 5))))
                attr = lstData.List(i, 0)
                siz = lstData.List(i, 1)
                If lots <> "" Then lots = lots & "、"
                lots = lots & lstData.List(i, 2)
                qty = qty + Val(lstData.List(i, 3))
                nLot = nLot + 1
            End If
        Next i

        topR = r

        If ONE_PER_PAGE And Not USE_LABEL_PRINTER And g > 1 Then
            sh.HPageBreaks.Add Before:=sh.Cells(topR, 1)
        End If

        If boxes.Count > 1 Then
            WriteRow sh, r, "箱番号", boxNo & " / " & boxes.Count, 18: r = r + 1
        End If
        WriteRow sh, r, "材質", attr, 16: r = r + 1
        WriteRow sh, r, "寸法", siz, 16: r = r + 1
        WriteRow sh, r, "ロット番号", lots, 12: r = r + 1
        WriteRow sh, r, "数量", Num(qty) & " 本（" & nLot & " ロット）", 16: r = r + 1

        With sh.Range("A" & topR & ":B" & (r - 1))
            .Borders(xlEdgeTop).Weight = xlMedium
            .Borders(xlEdgeBottom).Weight = xlMedium
            .Borders(xlEdgeLeft).Weight = xlMedium
            .Borders(xlEdgeRight).Weight = xlMedium
        End With

        bTop(g) = topR
        bBot(g) = r - 1
        r = r + 1

    Next g

    Application.ScreenUpdating = True

    On Error GoTo PrintErr

    If USE_LABEL_PRINTER Then
        SetupPage sh, True
        For g = 1 To order.Count
            sh.PageSetup.PrintArea = "$A$" & bTop(g) & ":$B$" & bBot(g)
            DoPrint sh
        Next g
    Else
        SetupPage sh, False
        sh.PageSetup.PrintArea = "$A$1:$B$" & (r - 1)
        DoPrint sh
    End If

    BuildAndPrint = True

PrintErr:
    Application.ScreenUpdating = True
    On Error Resume Next
    Application.ActivePrinter = oldPrinter
    On Error Resume Next
    sh.Visible = xlSheetHidden

End Function

Private Sub WriteRow(ByVal sh As Worksheet, ByVal r As Long, _
                     ByVal cap As String, ByVal txt As String, ByVal fs As Single)

    With sh.Cells(r, "A")
        .Value = cap
        .Font.Size = 14
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .Interior.Color = CLR_SECONDARY
    End With

    With sh.Cells(r, "B")
        .NumberFormat = "@"
        .Value = txt
        .Font.Size = fs
        .Font.Bold = (cap <> "ロット番号")
        .WrapText = True
        .HorizontalAlignment = xlLeft
        .VerticalAlignment = xlCenter
        .IndentLevel = 1
    End With

    With sh.Range("A" & r & ":B" & r).Borders
        .LineStyle = xlContinuous
        .Weight = xlThin
    End With

    sh.Rows(r).AutoFit
    If sh.Rows(r).RowHeight < 32 Then sh.Rows(r).RowHeight = 32

End Sub

Private Sub SetupPage(ByVal sh As Worksheet, ByVal isLabel As Boolean)

    On Error Resume Next

    With sh.PageSetup
        .Zoom = False
        .FitToPagesWide = 1
        .CenterHorizontally = True
        .CenterFooter = ""
        .LeftHeader = "": .CenterHeader = "": .RightHeader = ""

        If isLabel Then
            .FitToPagesTall = 1
            .CenterVertically = True
            .LeftMargin = Application.CentimetersToPoints(LABEL_MARGIN_CM)
            .RightMargin = Application.CentimetersToPoints(LABEL_MARGIN_CM)
            .TopMargin = Application.CentimetersToPoints(LABEL_MARGIN_CM)
            .BottomMargin = Application.CentimetersToPoints(LABEL_MARGIN_CM)
            .HeaderMargin = 0
            .FooterMargin = 0
            ' Kho giay: lay theo cai dat cua driver may in tem
            If LABEL_LANDSCAPE Then
                .Orientation = xlLandscape
            Else
                .Orientation = xlPortrait
            End If
        Else
            .FitToPagesTall = False
            .CenterVertically = False
            .Orientation = xlPortrait
            .PaperSize = xlPaperA4
        End If
    End With

End Sub

Private Sub DoPrint(ByVal sh As Worksheet)
    ' May in da duoc chon o BuildAndPrint
    sh.PrintOut
End Sub
