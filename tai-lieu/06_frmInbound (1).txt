Private Sub SetMode()

    Dim c As Object
    Dim isQR As Boolean

    isQR = optQR.Value

    For Each c In mQRCtl
        c.Visible = isQR
    Next c

    For Each c In mManual
        c.Visible = Not isQR
    Next c

    FocusInput

End Sub

    End If

End Sub

Private Sub LoadAttributes()

    Dim sh As Worksheet
    Dim r As Long
    Dim s As String

    cboAttr.Clear
    cboSize.Clear

    If chkAll.Value Then

        ' Tat ca hang trong マスタ
        Set sh = Sht(SH_MASTER)
        For r = 2 To LastRowOf(sh, "B")
            s = Trim$(CStr(sh.Cells(r, "B").Value))
            If s <> "" And Not ComboHas(cboAttr, s) Then cboAttr.AddItem s
        Next r

    Else

        ' Chi hang dang con ton kho (在庫 cot H > 0)
        Set sh = Sht(SH_STOCK)
        For r = 2 To LastRowOf(sh, "A")
            s = Trim$(CStr(sh.Cells(r, "A").Value))
            If s <> "" And NumOf(sh.Cells(r, "H").Value) > 0 Then
                If Not ComboHas(cboAttr, s) Then cboAttr.AddItem s
            End If
        Next r

    End If

End Sub

Private Sub chkAll_Click()
    LoadAttributes
    txtQty.Value = vbNullString
End Sub

Private Sub cboAttr_Change()

    Dim sh As Worksheet
    Dim r As Long
    Dim attr As String
    Dim s As String

    cboSize.Clear
    txtQty.Value = vbNullString

    attr = Trim$(CStr(cboAttr.Value))
    If attr = "" Then Exit Sub

    If chkAll.Value Then

        Set sh = Sht(SH_MASTER)
        For r = 2 To LastRowOf(sh, "B")
            If StrComp(Trim$(CStr(sh.Cells(r, "B").Value)), attr, vbTextCompare) = 0 Then
                s = Trim$(CStr(sh.Cells(r, "C").Value))
                If s <> "" And Not ComboHas(cboSize, s) Then cboSize.AddItem s
            End If
        Next r

    Else

        Set sh = Sht(SH_STOCK)
        For r = 2 To LastRowOf(sh, "A")
            If StrComp(Trim$(CStr(sh.Cells(r, "A").Value)), attr, vbTextCompare) = 0 _
            And NumOf(sh.Cells(r, "H").Value) > 0 Then
                s = Trim$(CStr(sh.Cells(r, "B").Value))
                If s <> "" And Not ComboHas(cboSize, s) Then cboSize.AddItem s
            End If
        Next r

    End If

End Sub

Private Sub cboSize_Change()

    Dim u As String
    Dim units As Double

    If GetMasterInfo(CStr(cboAttr.Value), CStr(cboSize.Value), u, units) Then
        txtQty.Value = Num(units)
    Else
        txtQty.Value = vbNullString
    End If

End Sub

Private Sub ProcessQR()

    Dim raw As String
    Dim attr As String, siz As String, lot As String
    Dim unitName As String
    Dim units As Double

    raw = CleanText(CStr(txtQR.Value))
    txtQR.Value = vbNullString

    If raw = "" Then Exit Sub

    If Not ParseGoodsQR(raw, attr, siz, lot) Then
        UI_Msg lblMsg, "QRコードの項目数が不足しています。商品QRを読んでください。", MSG_ERROR
        FocusInput
        Exit Sub
    End If

    If attr = "" Or siz = "" Or lot = "" Then
        UI_Msg lblMsg, "QRコードから属性・サイズ・ロット番号を取得できません。", MSG_ERROR
        FocusInput
        Exit Sub
    End If

    If Not GetMasterInfo(attr, siz, unitName, units) Then
        UI_Msg lblMsg, "マスタに該当品目がありません（" & attr & " / " & siz & "）", MSG_ERROR
        FocusInput
        Exit Sub
    End If

    If IsLotDuplicate(lot) Then
        UI_Msg lblMsg, "このロット番号は既に登録されています: " & lot, MSG_ERROR
        FocusInput
        Exit Sub
    End If

    AddListRow attr, siz, lot, units, unitName, units
    UI_Msg lblMsg, "OK  " & attr & " / " & siz & " / " & lot & "（" & Num(units) & " 本）", MSG_OK
    FocusInput

End Sub

Private Sub AddManual()

    Dim attr As String, siz As String, lot As String
    Dim unitName As String
    Dim units As Double
    Dim qty As Double

    attr = Trim$(CStr(cboAttr.Value))
    siz = Trim$(CStr(cboSize.Value))
    lot = NormalizeLotText(CStr(txtLot.Value))

    If attr = "" Then
        UI_Msg lblMsg, "属性を選択してください。", MSG_ERROR
        Exit Sub
    End If

    If siz = "" Then
        UI_Msg lblMsg, "サイズを選択してください。", MSG_ERROR
        Exit Sub
    End If

    If Not GetMasterInfo(attr, siz, unitName, units) Then
        UI_Msg lblMsg, "マスタに有効な 1QR本数 が登録されていません。", MSG_ERROR
        Exit Sub
    End If

    If Not IsNumeric(txtQty.Value) Then
        UI_Msg lblMsg, "本数を数字で入力してください。", MSG_ERROR
        Exit Sub
    End If

    qty = CDbl(txtQty.Value)

    If qty <= 0 Then
        UI_Msg lblMsg, "本数は 0 より大きい値を入力してください。", MSG_ERROR
        Exit Sub
    End If

    ' Khong co lo -> tu tao so NQ
    If lot = "" Then
        lot = MakeNoQRLot()
        AddListRow attr, siz, lot, qty, unitName, units
        UI_Msg lblMsg, "QRなしで追加しました: " & lot & "（" & Num(qty) & " 本）", MSG_OK
        FocusInput
        Exit Sub
    End If

    ' Khoang lo
    If InStr(1, lot, "~") > 0 Then
        If AddLotRange(lot, attr, siz, unitName, units, qty) Then
            txtLot.Value = vbNullString
        End If
        FocusInput
        Exit Sub
    End If

    If IsLotDuplicate(lot) Then
        UI_Msg lblMsg, "このロット番号は既に登録されています: " & lot, MSG_ERROR
        FocusInput
        Exit Sub
    End If

    AddListRow attr, siz, lot, qty, unitName, units
    txtLot.Value = vbNullString
    UI_Msg lblMsg, "OK  " & attr & " / " & siz & " / " & lot & "（" & Num(qty) & " 本）", MSG_OK
    FocusInput

End Sub

' H12345-1~10 -> H12345-1 ... H12345-10 (giu so chu so cua so dau)
Private Function AddLotRange( _
    ByVal s As String, ByVal attr As String, ByVal siz As String, _
    ByVal unitName As String, ByVal units As Double, ByVal qty As Double) As Boolean

    Dim leftPart As String, prefix As String
    Dim firstText As String, lastText As String
    Dim p As Long, dashPos As Long
    Dim firstNo As Long, lastNo As Long, digits As Long
    Dim i As Long
    Dim lot As String

    p = InStr(1, s, "~")
    leftPart = Left$(s, p - 1)
    lastText = Trim$(Mid$(s, p + 1))
    dashPos = InStrRev(leftPart, "-")

    If dashPos <= 1 Then GoTo Invalid

    prefix = Trim$(Left$(leftPart, dashPos - 1))
    firstText = Trim$(Mid$(leftPart, dashPos + 1))

    If prefix = "" Or firstText = "" Or lastText = "" Then GoTo Invalid
    If Not IsAllDigits(firstText) Or Not IsAllDigits(lastText) Then GoTo Invalid

    firstNo = CLng(firstText)
    lastNo = CLng(lastText)
    digits = Len(firstText)

    If lastNo < firstNo Then GoTo Invalid

    If lastNo - firstNo > 500 Then
        UI_Msg lblMsg, "一度に登録できるロット数が多すぎます（最大 501）。", MSG_ERROR
        Exit Function
    End If

    For i = firstNo To lastNo
        lot = prefix & "-" & Format$(i, String$(digits, "0"))
        If IsLotDuplicate(lot) Then
            UI_Msg lblMsg, "登録済みのロット番号があります: " & lot, MSG_ERROR
            Exit Function
        End If
    Next i

    For i = firstNo To lastNo
        lot = prefix & "-" & Format$(i, String$(digits, "0"))
        AddListRow attr, siz, lot, qty, unitName, units
    Next i

    UI_Msg lblMsg, (lastNo - firstNo + 1) & " ロットを追加しました（" & _
                   prefix & "-" & firstText & " ～ " & lastText & "）", MSG_OK
    AddLotRange = True
    Exit Function

Invalid:
    UI_Msg lblMsg, "ロット番号の形式が正しくありません。入力例: H12345-1~10", MSG_ERROR

End Function

Private Function IsAllDigits(ByVal s As String) As Boolean

    Dim i As Long

    If s = "" Then Exit Function

    For i = 1 To Len(s)
        If Mid$(s, i, 1) < "0" Or Mid$(s, i, 1) > "9" Then Exit Function
    Next i

    IsAllDigits = True

End Function

' NQ-yyyymmdd-001, 002, ... (khong trung)
Private Function MakeNoQRLot() As String

    Dim n As Long
    Dim lot As String
    Dim head As String

    head = "NQ-" & Format$(Date, "yyyymmdd") & "-"

    For n = 1 To 9999
        lot = head & Format$(n, "000")
        If Not IsLotDuplicate(lot) Then
            MakeNoQRLot = lot
            Exit Function
        End If
    Next n

End Function

Private Sub AddListRow( _
    ByVal attr As String, ByVal siz As String, ByVal lot As String, _
    ByVal qty As Double, ByVal unitName As String, ByVal units As Double)

    Dim k As Long

    lstDetails.AddItem Format$(Date, "yyyy/mm/dd")
    k = lstDetails.ListCount - 1

    lstDetails.List(k, 1) = attr
    lstDetails.List(k, 2) = siz
    lstDetails.List(k, 3) = lot
    lstDetails.List(k, 4) = Num(qty)
    lstDetails.List(k, 5) = unitName
    lstDetails.List(k, 6) = Num(units)

    lstDetails.ListIndex = k
    UpdateCount

End Sub

Private Function IsLotDuplicate(ByVal lot As String) As Boolean

    Dim i As Long

    lot = Trim$(lot)

    If lot = "" Then
        IsLotDuplicate = True
        Exit Function
    End If

    For i = 0 To lstDetails.ListCount - 1
        If StrComp(Trim$(CStr(lstDetails.List(i, 3))), lot, vbTextCompare) = 0 Then
            IsLotDuplicate = True
            Exit Function
        End If
    Next i

    IsLotDuplicate = LotUsed(lot)

End Function

Private Sub UpdateCount()

    Dim i As Long
    Dim total As Double

    For i = 0 To lstDetails.ListCount - 1
        total = total + Val(lstDetails.List(i, 4))
    Next i

    lblCount.Caption = "明細 " & lstDetails.ListCount & " 件　／　合計 " & Num(total) & " 本"

End Sub

Private Sub lstDetails_DblClick(ByVal Cancel As MSForms.ReturnBoolean)

    Dim i As Long

    i = lstDetails.ListIndex
    If i < 0 Then Exit Sub

    If MsgBox("この明細を削除しますか？" & vbCrLf & lstDetails.List(i, 3), _
              vbYesNo + vbQuestion, APP_NAME) = vbYes Then
        lstDetails.RemoveItem i
        UpdateCount
        UI_Msg lblMsg, "1 行削除しました。", MSG_INFO
    End If

    FocusInput

End Sub

Private Sub cmdRegister_Click()

    Dim slipNo As String
    Dim note As String
    Dim i As Long
    Dim lot As String
    Dim attr As String, siz As String
    Dim qty As Double
    Dim errMsg As String

    slipNo = Trim$(CStr(txtSlip.Value))

    If slipNo = "" Then
        UI_Msg lblMsg, "入庫伝票番号を必ず入力してください。", MSG_ERROR
        txtSlip.SetFocus
        Exit Sub
    End If

    If lstDetails.ListCount = 0 Then
        UI_Msg lblMsg, "入庫明細がありません。", MSG_ERROR
        Exit Sub
    End If

    ' Kiem tra truoc khi ghi
    For i = 0 To lstDetails.ListCount - 1

        lot = Trim$(CStr(lstDetails.List(i, 3)))

        If lot = "" Then
            UI_Msg lblMsg, "ロット番号が空欄の明細があります。", MSG_ERROR
            Exit Sub
        End If

        If LotUsed(lot) Then
            UI_Msg lblMsg, "登録済みのロット番号があります: " & lot, MSG_ERROR
            Exit Sub
        End If

        If Val(lstDetails.List(i, 4)) <= 0 Then
            UI_Msg lblMsg, "数量が正しくありません: " & lot, MSG_ERROR
            Exit Sub
        End If

    Next i

    If MsgBox("入庫を登録します。" & vbCrLf & vbCrLf & _
              "入庫伝票番号: " & slipNo & vbCrLf & _
              lblCount.Caption & vbCrLf & vbCrLf & _
              "よろしいですか？", vbYesNo + vbQuestion, APP_NAME) <> vbYes Then Exit Sub

    note = Trim$(CStr(txtNote.Value))

    On Error GoTo RegisterError
    Application.ScreenUpdating = False

    For i = 0 To lstDetails.ListCount - 1

        attr = CStr(lstDetails.List(i, 1))
        siz = CStr(lstDetails.List(i, 2))
        lot = CStr(lstDetails.List(i, 3))
        qty = Val(lstDetails.List(i, 4))

        WriteTrans slipNo, Date, K_IN, attr, siz, lot, qty, note

        errMsg = AdjustStock(attr, siz, qty, 0, False, True)
        If errMsg <> "" Then Err.Raise vbObjectError + 1, , errMsg

    Next i

    Application.ScreenUpdating = True

    lstDetails.Clear
    txtSlip.Value = vbNullString
    txtNote.Value = vbNullString
    txtLot.Value = vbNullString
    UpdateCount
    LoadAttributes

    UI_Msg lblMsg, "入庫登録が完了しました（伝票 " & slipNo & "）。", MSG_OK
    UI_Sound SOUND_DONE
    UI_Status Me, "入庫登録: " & slipNo & "  " & Format$(Now, "hh:mm"), MSG_OK
    txtSlip.SetFocus
    Exit Sub

RegisterError:
    Application.ScreenUpdating = True
    MsgBox "入庫登録中にエラーが発生しました。" & vbCrLf & Err.Description & vbCrLf & _
           "入出庫記録と在庫を確認してください。", vbCritical, APP_NAME

End Sub
