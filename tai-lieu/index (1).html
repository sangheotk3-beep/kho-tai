Public Sub UI_Sound(ByVal kind As Long)

    Dim f As String
    Dim sysAlias As String

    If Not SOUND_ON Then Exit Sub

    On Error GoTo Fallback

    Select Case kind
        Case SOUND_OK
            f = "Windows Navigation Start.wav"
            sysAlias = "SystemAsterisk"
        Case SOUND_ERROR
            f = "Windows Critical Stop.wav"
            sysAlias = "SystemHand"
        Case SOUND_DONE
            f = "tada.wav"
            sysAlias = "SystemExclamation"
        Case SOUND_WARN
            f = "Windows Exclamation.wav"
            sysAlias = "SystemQuestion"
        Case Else
            Exit Sub
    End Select

    f = Environ$("WINDIR") & "\Media\" & f

    If Dir(f) <> "" Then
        PlaySound f, 0, SND_ASYNC Or SND_FILENAME Or SND_NODEFAULT
    Else
        PlaySound sysAlias, 0, SND_ASYNC Or SND_ALIAS
    End If

    Exit Sub

Fallback:
    Beep

End Sub

    Next i

    s = Replace(s, ChrW(&H301C), "~")    ' wave dash
    s = Replace(s, ChrW(&HFF0D), "-")    ' fullwidth minus
    s = Replace(s, ChrW(&H2212), "-")    ' minus sign
    s = Replace(s, ChrW(&H2010), "-")    ' hyphen
    s = Replace(s, ChrW(&H2013), "-")    ' en dash
    s = Replace(s, ChrW(&H30FC), "-")    ' katakana long mark

    f = NumOf(sh.Cells(r, "F").Value) + dIn
    g = NumOf(sh.Cells(r, "G").Value) + dOut

                      Num(NumOf(sh.Cells(r, "H").Value)) & " 本）"
        Exit Function
    End If

    sh.Cells(r, "E").Value = e
    sh.Cells(r, "F").Value = f
    sh.Cells(r, "G").Value = g
    sh.Cells(r, "H").Value = h

End Function

Public Function SyncOpeningRow(ByVal r As Long) As Boolean

    Dim shM As Worksheet
    Dim shS As Worksheet
    Dim attr As String, siz As String
    Dim v As Variant
    Dim opening As Double
    Dim sr As Long

    Set shM = Sht(SH_MASTER)
    Set shS = Sht(SH_STOCK)

    attr = Trim$(CStr(shM.Cells(r, "B").Value))
    siz = Trim$(CStr(shM.Cells(r, "C").Value))
    v = shM.Cells(r, "F").Value

    If attr = "" Or siz = "" Then Exit Function

    If Trim$(CStr(v)) = "" Then
        opening = 0
    ElseIf IsNumeric(v) Then
        opening = CDbl(v)
    Else
        Exit Function
    End If

    sr = FindStockRow(attr, siz)

    If sr = 0 Then
        If opening = 0 Then Exit Function
        sr = NextRowOf(shS, "A")
        shS.Cells(sr, "A").Value = attr
        shS.Cells(sr, "B").Value = siz
        InvalidateStockCache
        shS.Cells(sr, "F").Value = 0
        shS.Cells(sr, "G").Value = 0
    End If

    shS.Cells(sr, "C").Value = shM.Cells(r, "D").Value
    shS.Cells(sr, "D").Value = shM.Cells(r, "E").Value
    shS.Cells(sr, "E").Value = opening
    shS.Cells(sr, "H").Value = opening + NumOf(shS.Cells(sr, "F").Value) - NumOf(shS.Cells(sr, "G").Value)

    SyncOpeningRow = True

End Function

' Tach CSV thanh cac dong; moi dong la mang chuoi (0-based). Ho tro "..." co dau phay.
Public Function ParseCsv(ByVal txt As String, ByVal delim As String) As Collection

    Dim csvRows As New Collection
    Dim fields() As String
    Dim nF As Long
    Dim cur As String
    Dim inQ As Boolean
    Dim i As Long
    Dim ch As String

    txt = Replace(txt, vbCrLf, vbLf)
    txt = Replace(txt, vbCr, vbLf)
    If Right$(txt, 1) <> vbLf Then txt = txt & vbLf

    ReDim fields(0 To 0)
    nF = 0

    For i = 1 To Len(txt)

        ch = Mid$(txt, i, 1)

        If inQ Then
            If ch = """" Then
                If Mid$(txt, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQ = False
                End If
            Else
                cur = cur & ch
            End If
        Else
            Select Case ch
                Case """"
                    inQ = True
                Case delim
                    ReDim Preserve fields(0 To nF)
                    fields(nF) = Trim$(cur)
                    nF = nF + 1
                    cur = vbNullString
                Case vbLf
                    ReDim Preserve fields(0 To nF)
                    fields(nF) = Trim$(cur)
                    csvRows.Add fields
                    nF = 0
                    cur = vbNullString
                    ReDim fields(0 To 0)
                Case Else
                    cur = cur & ch
            End Select
        End If

    Next i

    Set ParseCsv = csvRows

End Function

Private Sub CleanupBackups(ByVal folder As String)

    Dim files() As String
    Dim n As Long
    Dim f As String
    Dim i As Long, j As Long
    Dim t As String

    f = Dir(folder & Application.PathSeparator & "ストレート管理_*.*")

    Do While f <> ""
        ReDim Preserve files(0 To n)
        files(n) = f
        n = n + 1
        f = Dir()
    Loop

    If n <= BACKUP_KEEP Then Exit Sub

    ' Sap xep ten (ten co ngay gio -> cu truoc)
    For i = 0 To n - 2
        For j = i + 1 To n - 1
            If files(j) < files(i) Then
                t = files(i): files(i) = files(j): files(j) = t
            End If
        Next j
    Next i

    On Error Resume Next
    For i = 0 To n - BACKUP_KEEP - 1
        Kill folder & Application.PathSeparator & files(i)
    Next i

End Sub

Private Function CopySheetData(ByVal wbOld As Workbook, ByVal sheetName As String) As String

    Const MAX_COL As Long = 10

    Dim src As Worksheet
    Dim dst As Worksheet
    Dim lastRow As Long
    Dim c As Long
    Dim n As Long

    On Error Resume Next
    Set src = wbOld.Worksheets(sheetName)
    On Error GoTo 0

    If src Is Nothing Then
        CopySheetData = "・" & sheetName & " : 古いファイルにありません（スキップ）"
        Exit Function
    End If

    Set dst = GetOrCreateSheet(sheetName)

    If src.AutoFilterMode Then src.AutoFilterMode = False
    If dst.AutoFilterMode Then dst.AutoFilterMode = False

    For c = 1 To MAX_COL
        If LastRowOf(src, ColLetter(c)) > lastRow Then lastRow = LastRowOf(src, ColLetter(c))
    Next c

    ' Xoa du lieu cu cua file nay
    dst.Range(dst.Cells(2, 1), dst.Cells(dst.Rows.Count, MAX_COL)).ClearContents

    If lastRow < 2 Then
        CopySheetData = "・" & sheetName & " : 0 行"
        Exit Function
    End If

    n = lastRow - 1

    ' Cot ma: dinh dang chu truoc khi chep (giu so 0 o dau)
    Select Case sheetName
        Case SH_TRANS
            dst.Columns("B").NumberFormat = "@"
            dst.Columns("G").NumberFormat = "@"
        Case SH_DETAIL
            dst.Columns("B").NumberFormat = "@"
        Case SH_LOG
            dst.Columns("C").NumberFormat = "@"
            dst.Columns("D").NumberFormat = "@"
    End Select

    dst.Range("A2").Resize(n, MAX_COL).Value = src.Range("A2").Resize(n, MAX_COL).Value

    CopySheetData = "・" & sheetName & " : " & n & " 行"

End Function

Private Function AddLotRange( _
    ByVal s As String, ByVal attr As String, ByVal siz As String, _
    ByVal unitName As String, ByVal units As Double, ByVal qty As Double) As Boolean

    Dim leftPart As String, prefix As String
    Dim firstText As String, lastText As String
    Dim p As Long, dashPos As Long
    Dim firstNo As Long, lastNo As Long, digits As Long
    Dim i As Long
    Dim lot As String

    p = InStr(1, s, "~")
    leftPart = Left$(s, p - 1)
    lastText = Trim$(Mid$(s, p + 1))
    dashPos = InStrRev(leftPart, "-")

    If dashPos <= 1 Then GoTo Invalid

    prefix = Trim$(Left$(leftPart, dashPos - 1))
    firstText = Trim$(Mid$(leftPart, dashPos + 1))

    If prefix = "" Or firstText = "" Or lastText = "" Then GoTo Invalid
    If Not IsAllDigits(firstText) Or Not IsAllDigits(lastText) Then GoTo Invalid

    firstNo = CLng(firstText)
    lastNo = CLng(lastText)
    digits = Len(firstText)

    If lastNo < firstNo Then GoTo Invalid

    If lastNo - firstNo > 500 Then
        UI_Msg lblMsg, "一度に登録できるロット数が多すぎます（最大 501）。", MSG_ERROR
        Exit Function
    End If

    For i = firstNo To lastNo
        lot = prefix & "-" & Format$(i, String$(digits, "0"))
        If IsLotDuplicate(lot) Then
            UI_Msg lblMsg, "登録済みのロット番号があります: " & lot, MSG_ERROR
            Exit Function
        End If
    Next i

    For i = firstNo To lastNo
        lot = prefix & "-" & Format$(i, String$(digits, "0"))
        AddListRow attr, siz, lot, qty, unitName, units
    Next i

    UI_Msg lblMsg, (lastNo - firstNo + 1) & " ロットを追加しました（" & _
                   prefix & "-" & firstText & " ～ " & lastText & "）", MSG_OK
    AddLotRange = True
    Exit Function

Invalid:
    UI_Msg lblMsg, "ロット番号の形式が正しくありません。入力例: H12345-1~10", MSG_ERROR

End Function

Private Function IsAllDigits(ByVal s As String) As Boolean

    Dim i As Long

    Next i

End Function

        txtQR.SetFocus
        Exit Sub
    End If

    If attr = "" Or siz = "" Then
        UI_Msg lblMsg, "属性とサイズを選択してください。", MSG_ERROR

    End If

    If Not IsNumeric(txtQty.Value) Then

    End If

    If Not 品目存在(attr, siz) Then
        UI_Msg lblMsg, "マスタに登録されていない品目です。", MSG_ERROR
        Exit Sub
    End If

    stock = StockOf(attr, siz)

    If stock <= 0 Then
        UI_Msg lblMsg, "この品目は在庫がありません。", MSG_ERROR
        Exit Sub
    End If

    avail = stock - AddedQty(attr, siz)

    If qty > avail + EPS Then
        UI_Msg lblMsg, "在庫数を超えています（在庫 " & Num(stock) & " 本 / 選択可能 " & _
                       Num(avail) & " 本）", MSG_ERROR

    k = lstItems.ListCount - 1
    lstItems.List(k, 1) = CStr(LineCountOf(slip))
    lstItems.List(k, 2) = attr
    lstItems.List(k, 3) = siz

    UI_Msg lblMsg, "追加しました: " & attr & " / " & siz & "　" & Num(qty) & " 本", MSG_OK

Private Sub cmdSave_Click()

    Dim sh As Worksheet
    Dim r As Long
    Dim i As Long
    Dim slip As String
    Dim note As String
    Dim shipDate As Date

    If Not IsDate(txtShipDate.Value) Then
        UI_Msg lblMsg, "出庫日を正しく入力してください（例 2026/10/31）。", MSG_ERROR
        txtShipDate.SetFocus
        Exit Sub
    End If

    shipDate = CDate(txtShipDate.Value)

    If lstItems.ListCount = 0 Then
        UI_Msg lblMsg, "明細を 1 件以上追加してください。", MSG_ERROR
        Exit Sub
    End If

    If Int(shipDate) < Int(Date) Then
        If MsgBox("出庫日が過去の日付です。このまま登録しますか？", _
                  vbYesNo + vbQuestion, APP_NAME) <> vbYes Then Exit Sub
    End If

    For i = 0 To lstItems.ListCount - 1
        If AddedQty(CStr(lstItems.List(i, 2)), CStr(lstItems.List(i, 3))) > _
           StockOf(CStr(lstItems.List(i, 2)), CStr(lstItems.List(i, 3))) + EPS Then
            UI_Msg lblMsg, "在庫が不足している明細があります: " & _
                           lstItems.List(i, 2) & " / " & lstItems.List(i, 3), MSG_ERROR
            Exit Sub
        End If
    Next i

    Set sh = Sht(SH_DETAIL)

    For i = 0 To lstItems.ListCount - 1

        slip = CStr(lstItems.List(i, 0))

        If slip = "" Then
            UI_Msg lblMsg, "伝票番号が空欄の明細があります。", MSG_ERROR
            Exit Sub
        End If

        If SlipRegistered(slip) Then
            UI_Msg lblMsg, "この伝票QRコードは既に登録されています: " & slip, MSG_ERROR
            Exit Sub
        End If

    Next i

    If MsgBox("出庫伝票を登録します。" & vbCrLf & vbCrLf & _
              "出庫日: " & Format$(shipDate, "yyyy/mm/dd") & vbCrLf & _
              "伝票数: " & SlipCount() & vbCrLf & _
              "明細数: " & lstItems.ListCount & " 行" & vbCrLf & vbCrLf & _
              "この時点では在庫は減りません。よろしいですか？", _
              vbYesNo + vbQuestion, APP_NAME) <> vbYes Then Exit Sub

    note = Trim$(CStr(txtNote.Value))

    On Error GoTo SaveError
    Application.ScreenUpdating = False

    For i = 0 To lstItems.ListCount - 1

        r = NextRowOf(sh, "A")

        sh.Cells(r, "A").Value = r - 1
        sh.Cells(r, "B").NumberFormat = "@"
        sh.Cells(r, "B").Value = CStr(lstItems.List(i, 0))
        sh.Cells(r, "C").Value = lstItems.List(i, 2)
        sh.Cells(r, "D").Value = lstItems.List(i, 3)
        sh.Cells(r, "E").Value = Val(lstItems.List(i, 4))
        sh.Cells(r, "F").Value = ST_PENDING
        sh.Cells(r, "G").Value = note
        sh.Cells(r, "H").NumberFormat = "yyyy/mm/dd"
        sh.Cells(r, "H").Value = shipDate
        sh.Cells(r, "J").Value = CLng(lstItems.List(i, 1))

    Next i

    Application.ScreenUpdating = True

    UI_Msg lblMsg, "出庫伝票を登録しました（明細 " & lstItems.ListCount & " 行・未出庫）。", MSG_OK
    UI_Sound SOUND_DONE
    UI_Status Me, "登録: " & Format$(Now, "hh:mm") & "　在庫数は変更されていません。", MSG_OK

    ResetForm
    Exit Sub

SaveError:
    Application.ScreenUpdating = True
    MsgBox "登録中にエラーが発生しました。" & vbCrLf & Err.Description & vbCrLf & _
           "OutputDetails シートを確認してください。", vbCritical, APP_NAME

End Sub

Private Sub cmdCsv_Click()

    Dim f As Variant
    Dim csvRows As Collection
    Dim rw As Variant
    Dim i As Long
    Dim lineNo As Long
    Dim slip As String, attr As String, siz As String
    Dim qtyTxt As String
    Dim qty As Double
    Dim units As Double
    Dim avail As Double
    Dim k As Long
    Dim okN As Long, ngN As Long
    Dim errs As String
    Dim reason As String
    Dim firstDate As String, firstNote As String

    f = Application.GetOpenFilename("CSV ファイル (*.csv;*.txt),*.csv;*.txt", , "出庫伝票の CSV を選択")
    If VarType(f) = vbBoolean Then Exit Sub

    On Error GoTo ReadErr
    Set csvRows = ParseCsv(ReadTextFile(CStr(f), CSV_CHARSET), CSV_DELIM)
    On Error GoTo 0

    For Each rw In csvRows

        lineNo = lineNo + 1
        If lineNo <= CSV_HEADER_ROWS Then GoTo NextRow

        slip = QR伝票コード(CsvField(rw, CSV_COL_SLIP))
        attr = CsvField(rw, CSV_COL_ATTR)
        siz = CsvField(rw, CSV_COL_SIZE)
        qtyTxt = Replace(CsvField(rw, CSV_COL_QTY), ",", "")

        ' Bo dong trong hoan toan
        If slip = "" And attr = "" And siz = "" And qtyTxt = "" Then GoTo NextRow

        reason = vbNullString

        If slip = "" Then
            reason = "伝票番号が空欄"
        ElseIf SlipRegistered(slip) Then
            reason = "伝票番号 " & slip & " は登録済み"
        ElseIf attr = "" Or siz = "" Then
            reason = "属性・サイズが空欄"
        ElseIf Not 品目存在(attr, siz) Then
            reason = "マスタにない品目 (" & attr & " / " & siz & ")"
        ElseIf Not IsNumeric(qtyTxt) Then
            reason = "数量が数字ではありません (" & qtyTxt & ")"
        End If

        If reason = "" Then

            qty = CDbl(qtyTxt)

            If CSV_QTY_IS_LOTS Then
                If GetUnitsPerQR(attr, siz, units) Then
                    qty = qty * units
                Else
                    reason = "マスタに 1QR本数 がありません (" & attr & " / " & siz & ")"
                End If
            End If

        End If

        If reason = "" Then
            If qty <= 0 Then
                reason = "数量が 0 以下"
            Else
                avail = StockOf(attr, siz) - AddedQty(attr, siz)
                If qty > avail + EPS Then
                    reason = "在庫不足 (" & attr & " / " & siz & "　必要 " & Num(qty) & " / 選択可能 " & Num(avail) & ")"
                End If
            End If
        End If

        If reason <> "" Then
            ngN = ngN + 1
            If ngN <= 25 Then errs = errs & vbCrLf & "・" & lineNo & " 行目: " & reason
            GoTo NextRow
        End If

        lstItems.AddItem slip
        k = lstItems.ListCount - 1
        lstItems.List(k, 1) = CStr(LineCountOf(slip))
        lstItems.List(k, 2) = attr
        lstItems.List(k, 3) = siz
        lstItems.List(k, 4) = Num(qty)
        okN = okN + 1

        If firstDate = "" And CSV_COL_DATE > 0 Then firstDate = CsvField(rw, CSV_COL_DATE)
        If firstNote = "" And CSV_COL_NOTE > 0 Then firstNote = CsvField(rw, CSV_COL_NOTE)

NextRow:
    Next rw

    If IsDate(firstDate) Then txtShipDate.Value = Format$(CDate(firstDate), "yyyy/mm/dd")
    If firstNote <> "" And Trim$(CStr(txtNote.Value)) = "" Then txtNote.Value = firstNote

    UpdateCounts
    UpdateStockLabel

    If ngN = 0 Then
        UI_Msg lblMsg, "CSV から " & okN & " 行を読み込みました。内容を確認して「登録完了」を押してください。", MSG_OK
    Else
        UI_Msg lblMsg, "CSV: " & okN & " 行 OK / " & ngN & " 行 エラー（エラー行は追加していません）", MSG_ERROR
        MsgBox "次の行は追加できませんでした:" & errs & _
               IIf(ngN > 25, vbCrLf & "…ほか " & (ngN - 25) & " 行", ""), vbExclamation, APP_NAME
    End If

    Exit Sub

ReadErr:
    UI_Msg lblMsg, "CSV を読み込めませんでした: " & Err.Description, MSG_ERROR

End Sub

Private Sub LoadSlip(ByVal raw As String)

    Dim sh As Worksheet
    Dim r As Long
    Dim slip As String
    Dim found As Boolean
    Dim needQty As Double

    If Not IsDate(txtDate.Value) Then
        ShowError "出庫日を正しく入力してください（例 2026/10/31）。"
        Exit Sub
    End If

    slip = QR伝票コード(raw)
    If slip = "" Then FocusQR: Exit Sub

    Set sh = Sht(SH_DETAIL)
    mN = 0

    For r = 2 To LastRowOf(sh, "B")

        If StrComp(Trim$(CStr(sh.Cells(r, "B").Value)), slip, vbTextCompare) = 0 Then

            found = True

            If Trim$(CStr(sh.Cells(r, "F").Value)) = ST_PENDING Then

                needQty = NumOf(sh.Cells(r, "E").Value)

                If needQty > 0 Then

                    ReDim Preserve mRow(0 To mN)
                    ReDim Preserve mLine(0 To mN)
                    ReDim Preserve mAttr(0 To mN)
                    ReDim Preserve mSize(0 To mN)
                    ReDim Preserve mNeed(0 To mN)
                    ReDim Preserve mDone(0 To mN)

                    mRow(mN) = r
                    mLine(mN) = LineNoOf(sh, r, slip)
                    mAttr(mN) = Trim$(CStr(sh.Cells(r, "C").Value))
                    mSize(mN) = Trim$(CStr(sh.Cells(r, "D").Value))
                    mNeed(mN) = needQty
                    mDone(mN) = 0

                    If mNote = "" Then mNote = CStr(sh.Cells(r, "G").Value)

                    mN = mN + 1

                End If

            End If

        End If

    Next r

    If Not found Then
        mN = 0
        ShowError "この伝票は登録されていません: " & slip
        Exit Sub
    End If

    If mN = 0 Then
        ShowError "この伝票は既に出庫済み（または取消）です: " & slip
        Exit Sub
    End If

    mSlip = slip
    mCur = -1
    mS = 0

    txtSlipCode.Value = slip

    ShowSlipList
    UpdateProgress
    UI_Msg lblMsg, "伝票を読み込みました（" & mN & " 行）。商品QRまたはロット番号を読み取ってください。", MSG_OK
    FocusQR

Private Function LineNoOf(ByVal sh As Worksheet, ByVal r As Long, ByVal slip As String) As Long

    Dim k As Long

    For k = 2 To r
        If StrComp(Trim$(CStr(sh.Cells(k, "B").Value)), slip, vbTextCompare) = 0 Then
            LineNoOf = LineNoOf + 1
        End If
    Next k

End Function

                lstList.List(k, 5) = Trim$(CStr(sh.Cells(r, "D").Value))

                If GetUnitsPerQR(CStr(lstList.List(k, 4)), CStr(lstList.List(k, 5)), units) Then

Private Sub ProcessLot(ByVal raw As String)

    Dim lot As String
    Dim attr As String, siz As String
    Dim qrAttr As String, qrSize As String
    Dim qty As Double
    Dim state As Long
    Dim isQR As Boolean
    Dim isOld As Boolean
    Dim idx As Long
    Dim i As Long
    Dim recAttr As String, recSize As String
    Dim recQty As Double

    isQR = ParseGoodsQR(raw, qrAttr, qrSize, lot)
    If Not isQR Then lot = raw

    If lot = "" Then
        ShowError "ロット番号を取得できません。"
        Exit Sub
    End If

    For i = 0 To mS - 1
        If StrComp(sLot(i), lot, vbTextCompare) = 0 Then
            ShowError "このロットは既に読み取り済みです: " & lot
            Exit Sub
        End If
    Next i

    state = LotState(lot, recAttr, recSize, recQty)

    If state = LOT_OUT Then
        ShowError "このロットは既に出庫済みです: " & lot
        Exit Sub
    End If

    If isQR And state <> LOT_IN Then
        ShowError "このロットは入庫登録されていません: " & lot
        Exit Sub
    End If

    If state = LOT_IN Then

        isOld = False
        attr = recAttr
        siz = recSize

        If mCur >= 0 Then

            If StrComp(mAttr(mCur), attr, vbTextCompare) <> 0 _
            Or StrComp(mSize(mCur), siz, vbTextCompare) <> 0 Then
                ShowError "別の品目です。読み取り中の品目を完了してください（読取中: " & _
                          mAttr(mCur) & " " & mSize(mCur) & " ／ 今回: " & attr & " " & siz & "）"
                Exit Sub
            End If

            idx = mCur

        Else

            idx = FindLineByItem(attr, siz)

            If idx < 0 Then
                ShowError "この伝票にない品目、または既に完了しています: " & attr & " " & siz
                Exit Sub
            End If

        End If

    Else

        ' Hang cu khong QR
        isOld = True

        If mCur >= 0 Then
            idx = mCur
        ElseIf SelectedLine() >= 0 Then
            idx = SelectedLine()
        Else
            idx = OnlyOpenLine()
        End If

        If idx < 0 Then
            ShowError "旧在庫のロットです。リストの明細（品目）を選択してから入力してください。"
            Exit Sub
        End If

        attr = mAttr(idx)
        siz = mSize(idx)

    End If

    ' So ong 1 lo = 1QR本数 trong マスタ
    If Not GetUnitsPerQR(attr, siz, qty) Then
        If isOld Or recQty <= 0 Then
            ShowError "マスタに 1QR本数 が登録されていません: " & attr & " / " & siz
            Exit Sub
        End If
        qty = recQty
    End If

    If mNeed(idx) - mDone(idx) < qty - EPS Then
        ShowError "数量オーバーです（" & mAttr(idx) & " " & mSize(idx) & "　必要 " & _
                  Num(mNeed(idx)) & " / 読取済 " & Num(mDone(idx)) & " / このロット " & Num(qty) & " 本）"
        Exit Sub
    End If

    If FindStockRow(attr, siz) = 0 Then
        ShowError "在庫シートに該当品目がありません: " & attr & " / " & siz
        Exit Sub
    End If

    If StockOf(attr, siz) < ScannedQty(attr, siz) + qty - EPS Then
        ShowError "在庫が不足しています（" & attr & " / " & siz & "　在庫 " & _
                  Num(StockOf(attr, siz)) & " 本）"
        Exit Sub
    End If

    ReDim Preserve sLot(0 To mS)
    ReDim Preserve sAttr(0 To mS)
    ReDim Preserve sSize(0 To mS)
    ReDim Preserve sQty(0 To mS)
    ReDim Preserve sOld(0 To mS)

    sLot(mS) = lot
    sAttr(mS) = attr
    sSize(mS) = siz
    sQty(mS) = qty
    sOld(mS) = isOld
    mS = mS + 1

    mDone(idx) = mDone(idx) + qty
    mCur = idx

    If mDone(idx) >= mNeed(idx) - EPS Then
        mCur = -1
        ShowSlipList
        UI_Msg lblMsg, attr & " " & siz & " : 読み取り完了", MSG_OK
    Else
        ShowSlipList
        If isOld Then
            UI_Msg lblMsg, "OK（旧在庫）  " & lot & "　" & Num(qty) & " 本", MSG_OK
        Else
            UI_Msg lblMsg, "OK  " & lot & "　" & Num(qty) & " 本", MSG_OK
        End If
    End If

    UpdateProgress

    If AllDone() Then
        CommitSlip
    Else
        FocusQR
    End If

End Sub

Private Function AllDone() As Boolean

    Next i

        End If
    Next i

End Function

Private Function ScannedQty(ByVal attr As String, ByVal siz As String) As Double

    Dim i As Long

    For i = 0 To mS - 1
        If StrComp(sAttr(i), attr, vbTextCompare) = 0 _
        And StrComp(sSize(i), siz, vbTextCompare) = 0 Then
            ScannedQty = ScannedQty + sQty(i)
        End If
    Next i

End Function

        errMsg = AdjustStock(sAttr(i), sSize(i), 0, sQty(i))

    lotCount = mS

Private Sub ScanLot(ByVal lot As String)

    Dim sh As Worksheet
    Dim r As Long, lastRow As Long
    Dim inRow As Long, outRow As Long
    Dim kind As String
    Dim key As String
    Dim added As Long

    If lot = "" Then
        UI_Msg lblMsg, "ロット番号を取得できません。", MSG_ERROR
        Exit Sub
    End If

    Set sh = Sht(SH_TRANS)
    lastRow = LastRowOf(sh, "D")

    For r = 2 To lastRow
        If StrComp(Trim$(CStr(sh.Cells(r, "G").Value)), lot, vbTextCompare) = 0 Then
            kind = Trim$(CStr(sh.Cells(r, "D").Value))
            If kind = K_IN Then inRow = r
            If kind = K_OUT Then outRow = r
        End If
    Next r

    If outRow > 0 Then

        key = MakeKey(sh.Cells(outRow, "B").Value, sh.Cells(outRow, "E").Value, sh.Cells(outRow, "F").Value)

        For r = 2 To lastRow
            If Trim$(CStr(sh.Cells(r, "D").Value)) = K_OUT Then
                If MakeKey(sh.Cells(r, "B").Value, sh.Cells(r, "E").Value, sh.Cells(r, "F").Value) = key Then
                    added = added + AddTransRow(sh, r, K_OUT_CANCEL)
                End If
            End If
        Next r

        If added = 0 Then
            UI_Msg lblMsg, "ロット " & lot & " はすでに一覧にあります。", MSG_ERROR
        Else
            UI_Msg lblMsg, "出庫済み → 在庫へ戻します（伝票 " & sh.Cells(outRow, "B").Value & _
                           " / " & added & " ロット追加）", MSG_OK
        End If

    ElseIf inRow > 0 Then

        If AddTransRow(sh, inRow, K_IN_CANCEL) = 0 Then
            UI_Msg lblMsg, "ロット " & lot & " はすでに一覧にあります。", MSG_ERROR
        Else
            UI_Msg lblMsg, "入庫取消: ロット " & lot & " を追加しました。", MSG_OK
        End If

    Else

        UI_Msg lblMsg, "「" & lot & "」は取消できるデータがありません（未登録 または 取消済み）。", MSG_ERROR

    End If

Private Function CommitShip(ByVal slip As String) As Boolean

    Dim need As Object
    Dim i As Long
    Dim attr As String, siz As String, lot As String
    Dim qty As Double
    Dim key As Variant
    Dim note As String
    Dim errMsg As String

    Set need = CreateObject("Scripting.Dictionary")

        If LotState(lot) = LOT_OUT Then
            MsgBox "ロット " & lot & " はすでに出庫済みです。一覧から削除してください。", vbExclamation, APP_NAME
            Exit Function
        End If

        key = CStr(lstData.List(i, 0)) & vbTab & CStr(lstData.List(i, 1))
        need(key) = need(key) + Val(lstData.List(i, 3))

    Next i

    For Each key In need.keys
        attr = Split(key, vbTab)(0)
        siz = Split(key, vbTab)(1)
        If StockOf(attr, siz) < need(key) - EPS Then
            MsgBox "在庫が足りません: " & attr & " / " & siz & vbCrLf & _
                   "在庫 " & Num(StockOf(attr, siz)) & " 本 / 出庫 " & Num(need(key)) & " 本", _
                   vbExclamation, APP_NAME
            Exit Function
        End If
    Next key

    On Error GoTo CommitError
    Application.ScreenUpdating = False

    For i = 0 To lstData.ListCount - 1

        attr = CStr(lstData.List(i, 0))
        siz = CStr(lstData.List(i, 1))
        lot = CStr(lstData.List(i, 2))
        qty = Val(lstData.List(i, 3))

        If CStr(lstData.List(i, 4)) = "1" Then
            note = "熊本行き (旧在庫)"
        Else
            note = "熊本行き"
        End If

        WriteTrans slip, Date, K_OUT, attr, siz, lot, qty, note

        errMsg = AdjustStock(attr, siz, 0, qty)
        If errMsg <> "" Then Err.Raise vbObjectError + 1, , errMsg

    Next i

    Application.ScreenUpdating = True
    CommitShip = True
    Exit Function

CommitError:
    Application.ScreenUpdating = True
    MsgBox "出庫登録中にエラーが発生しました。" & vbCrLf & Err.Description, vbCritical, APP_NAME

End Function
